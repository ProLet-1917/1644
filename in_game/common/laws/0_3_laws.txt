#Swedish Empire
swe_imperial_parliament = {
    type = international_organization
    law_category = leadership

    potential = {
        international_organization_type = international_organization_type:swedish_empire
    }

    swe_empire_parliament_policy = {
        on_activate = {
            set_parliament_type = parliament_type:swe_empire_parliament
            change_variable = {
                name = swedish_empire_authority
                add = 0.02
            }
        }

        wants_propose_policy = {
            add = {
                desc = "POLICY_BIAS_LEADER_WANTS"
                value = 50
            }
        }

    }

    no_parliament_policy = {
        on_activate = {
            set_parliament_active = no
            change_variable = {
                name = swedish_empire_authority
                add = -0.1
            }
        }
    }

}
