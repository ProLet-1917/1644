declare_unification_of_germany_empire = {
    type = situation

    potential = {
        scope:actor = {
            has_variable = potential_unifier
        }
        situation:germany_unification = {
            situation_is_active = yes
        }
    }

    allow = {
        custom_tooltip = {
            text = only_1_unifier_now
            global_variable_list_size = {
                name = potential_unifier_list
                value <= 1
            }
        }
        scope:actor = {
            has_variable = potential_unifier
        }
    }

    ai_tick = monthly
    ai_tick_frequency = 3

    automation_tick = never
    automation_tick_frequency = 12

    select_trigger = {
        looking_for_a = situation
        target_flag = recipient
        name = "choose_situation"
        column = {data = name}
        visible = {
            situation:germany_unification = this
            situation_is_active = yes
        }
    }

    select_trigger = {
        looking_for_a = international_organization
        target_flag = target
        name = "choose_international_organization"
        column = {data = name}
        visible = {
            international_organization_type = international_organization_type:germany_unification_union
            scope:actor = scope:target.leader_country
        }
    }

    effect = {
        scope:actor = {
            form_country = formable_country:GER_f

        }
        scope:target = {
            every_international_organization_member = {
                trigger_event_non_silently = germany_unification.1001
            }
        }
    }

    ai_will_do = {
        value = 1000
    }
}

support_the_unifier = {
    type = situation

    potential = {
        scope:actor = {
            has_variable = normal_germany_country
        }
        situation:germany_unification = {
            situation_is_active = yes
        }
    }

    allow = {
        scope:actor = {
            has_variable = normal_germany_country
            NOT = {
                is_member_of_international_organization_of_type = {
                    type = germany_unification_union
                }
            }
        }
    }

    ai_tick = monthly
    ai_tick_frequency = 3

    automation_tick = never
    automation_tick_frequency = 12

    select_trigger = {
        looking_for_a = situation
        target_flag = recipient
        name = "choose_situation"
        column = {data = name}
        visible = {
            situation:germany_unification = this
            situation_is_active = yes
        }
    }

    select_trigger = {
        looking_for_a = international_organization
        target_flag = target
        name = "choose_international_organization"
        column = {data = name}
        visible = {
            international_organization_type = international_organization_type:germany_unification_union
        }
    }

    effect = {
        scope:target = {
            add_country_to_international_organization = scope:actor
        }
    }

    ai_will_do = {
        base = -25
        subtract = {
            desc = "game_concept_tax_base"
            value = {
                value = country_tax_base
                divide = 25
            }
        }
        add = {
            desc = "game_concept_opinion"
            value = {
                value = "root.opinion(scope:target.leader_country)"
                divide = 4      # max ±50 at ±200 opinion
            }
        }
        add = {
            desc = "game_concept_diplomatic_reputation"
            value = {
                value = "scope:target.leader_country.modifier:diplomatic_reputation"
                multiply = 0.5  # +0.5 reasons per 1 diprep
            }
        }
        if = {
            limit = {
                scope:target.leader_country = {
                    is_great_power = yes
                }
            }
            add = {
                desc = "game_concept_great_powers"
                value = 25
            }
        }
        if = {
            limit = {
                at_war = yes
            }
            subtract = {
                desc = "IS_AT_WAR"
                value = 1000
            }
        }
        if = {
            limit = {
                is_subject = yes
                NOT = { is_subject_of = international_organization:germany_unification_union.leader_country }
            }
            subtract = {
                desc = "IS_SUBJECT_TRIGGER"
                value = 1000
            }
        }
    }
}
