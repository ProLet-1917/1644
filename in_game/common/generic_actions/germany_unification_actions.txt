declare_unification_of_germany_empire = {
    type = situation

    potential = {
        scope:actor = {
            has_variable = potential_unifier
            NOT = { is_emperor = yes }
        }
        situation:germany_unification = {
            situation_is_active = yes
        }
    }

    allow = {
        custom_tooltip = {
            text = only_1_unifier_now
            global_variable_list_size = {
                name = potential_unifier_list
                value <= 1
            }
        }
        scope:actor = {
            has_variable = potential_unifier
        }
    }

    ai_tick = monthly
    ai_tick_frequency = 3

    automation_tick = never
    automation_tick_frequency = 12

    select_trigger = {
        looking_for_a = situation
        target_flag = recipient
        name = "choose_situation"
        column = {data = name}
        visible = {
            situation:germany_unification = this
            situation_is_active = yes
        }
    }

    select_trigger = {
        looking_for_a = international_organization
        target_flag = target
        name = "choose_international_organization"
        column = {data = name}
        visible = {
            international_organization_type = international_organization_type:germany_unification_union
            scope:actor = scope:target.leader_country
        }
    }

    effect = {
        scope:actor = {
            form_country = formable_country:GER_f
        }
        scope:target = {
            every_international_organization_member = {
                trigger_event_non_silently = germany_unification.1001
            }
        }
        every_country = {
            limit = {
                culture = { has_culture_group = culture_group:german_group }
                NOT = {
                    is_member_of_international_organization = scope:target
                }
            }
            trigger_event_non_silently = germany_unification.1002
        }
        every_international_organization = {
            limit = {
                international_organization_type = international_organization_type:germany_unification_union
            }
            destroy_international_organization = { target = this }
        }
    }

    ai_will_do = {
        value = 1000
    }
}

# 神罗仍在且本国为皇帝时，通过帝国统一德国
declare_unification_through_hre = {
    type = situation

    potential = {
        scope:actor = {
            culture = { has_culture_group = culture_group:german_group }
            is_emperor = yes
            NOT = { tag = GER }
        }
        exists = international_organization:hre
        situation:germany_unification = {
            situation_is_active = yes
        }
    }

    allow = {
        scope:actor = {
            is_emperor = yes
            has_advance = nation_state_advance
            is_subject = no
        }
        OR = {
            international_organization:hre = {
                international_organization_has_policy = policy:revoke_privilegia_policy
            }
            AND = {
                region:north_german_region = {
                    NOT = {
                        any_ownable_location_in_region = {
                            percent > 0.25
                            AND = {
                                NOT = { owner = scope:actor }
                                owner = {
                                    NOT = { is_subject_of = scope:actor }
                                }
                            }
                        }
                    }
                }
                region:south_german_region = {
                    NOT = {
                        any_ownable_location_in_region = {
                            percent > 0.35
                            AND = {
                                NOT = { owner = scope:actor }
                                owner = {
                                    NOT = { is_subject_of = scope:actor }
                                }
                            }
                        }
                    }
                }
            }
        }
    }

    ai_tick = monthly
    ai_tick_frequency = 3

    automation_tick = never
    automation_tick_frequency = 12

    select_trigger = {
        looking_for_a = situation
        target_flag = recipient
        name = "choose_situation"
        column = {data = name}
        visible = {
            situation:germany_unification = this
            situation_is_active = yes
        }
    }

    effect = {
        scope:actor = {
            save_scope_as = unifier
            form_country = formable_country:GER_f
        }
        international_organization:hre = {
            every_international_organization_member = {
                limit = { NOT = { this = scope:unifier } }
                trigger_event_non_silently = germany_unification.1101
            }
        }
        destroy_international_organization = { target = international_organization:hre }
        every_international_organization = {
            limit = {
                international_organization_type = international_organization_type:germany_unification_union
            }
            destroy_international_organization = { target = this }
        }
        every_country = {
            limit = {
                OR = {
                    has_variable = german_country
                    has_variable = potential_unifier
                }
            }
            remove_variable = german_country
            remove_variable = potential_unifier
        }
    }

    ai_will_do = {
        value = 1000
    }
}

support_the_unifier = {
    type = situation

    potential = {
        scope:actor = {
            has_variable = german_country
        }
        situation:germany_unification = {
            situation_is_active = yes
        }
    }

    allow = {
        scope:actor = {
            has_variable = german_country
            NOT = {
                is_member_of_international_organization_of_type = {
                    type = germany_unification_union
                }
            }
        }
    }

    ai_tick = monthly
    ai_tick_frequency = 3

    automation_tick = never
    automation_tick_frequency = 12

    select_trigger = {
        looking_for_a = situation
        target_flag = recipient
        name = "choose_situation"
        column = {data = name}
        visible = {
            situation:germany_unification = this
            situation_is_active = yes
        }
    }

    select_trigger = {
        looking_for_a = international_organization
        target_flag = target
        name = "choose_international_organization"
        column = {data = name}
        visible = {
            international_organization_type = international_organization_type:germany_unification_union
        }
    }

    effect = {
        scope:target = {
            add_country_to_international_organization = scope:actor
        }
    }

    ai_will_do = {
        base = -10
        subtract = {
            desc = "game_concept_tax_base"
            value = {
                value = country_tax_base
                divide = 25
            }
        }
        add = {
            desc = "game_concept_opinion"
            value = {
                value = "root.opinion(scope:target.leader_country)"
                divide = 4      # max ±50 at ±200 opinion
            }
        }
        add = {
            desc = "game_concept_diplomatic_reputation"
            value = {
                value = "scope:target.leader_country.modifier:diplomatic_reputation"
                multiply = 0.5  # +0.5 reasons per 1 diprep
            }
        }
        if = {
            limit = {
                scope:target.leader_country = {
                    is_great_power = yes
                }
            }
            add = {
                desc = "game_concept_great_powers"
                value = 25
            }
        }
        if = {
            limit = {
                at_war = yes
            }
            subtract = {
                desc = "IS_AT_WAR"
                value = 1000
            }
        }
        if = {
            limit = {
                is_subject = yes
                NOT = { is_subject_of = international_organization:germany_unification_union.leader_country }
            }
            subtract = {
                desc = "IS_SUBJECT_TRIGGER"
                value = 1000
            }
        }
    }
}
