organized_intervention_forces = {
    type = situation

    show_message = no
    potential = {
        scope:actor = {
            OR = {
                has_presence_in = region:ruthenia_region
                has_presence_in = region:baltic_region
                this = c:WAL
            }
            NOT = {
                country_rank_is_county = yes
                is_subject = yes
                this = c:PLC
                this = c:SWE
                this = c:RUS
                this = c:ZAZ
            }
        }
    }

    allow = {
        scope:actor = {
            NOT = {
                OR = {
                    is_at_war_with = c:PLC
                    is_at_war_with = c:ZAZ
                }
            }
        }
    }

    automation_tick = never
    automation_tick_frequency = 12
    ai_tick = monthly
    ai_tick_frequency = 6
    price = 1

    cooldown = {
        type = organized_intervention_forces_key
        years = 5
    }

    select_trigger = {
        looking_for_a = situation
        target_flag = recipient
        name = "choose_situation"
        column = {data = name}
        visible = {
            situation:the_deluge = this
            situation_is_active = yes
        }
    }

    select_trigger = {
        looking_for_a = character
        source = actor
        target_flag = target_character
        name = "choose_character"
        column = { data = name }
        visible = {
            is_female = no
            is_artist = no
            is_ruler = no
            is_heir = no
            is_child = no
            is_dynasty_head = no
        }
    }

    select_trigger = {
        looking_for_a = country
        target_flag = target
        name = "choose_support_country"
        column = { data = name}
        visible = {
            OR = {
                this = c:PLC
                this = {
                    is_at_war_with = c:PLC
                }
            }
        }
    }

    effect = {
        every_country = {
            limit = {
                is_at_war_with = scope:target
            }
            save_scope_as = target_2
        }
        scope:actor = {
            set_variable = {
                name = army_inf_scale
                value = scope:actor.monthly_income_total
            }
            set_variable = {
                name = army_cav_scale
                value = scope:actor.monthly_income_total
            }
            set_variable = {
                name = army_art_scale
                value = scope:actor.monthly_income_total
            }
            change_variable = {
                name = army_inf_scale
                multiply = 0.05
            }
            change_variable = {
                name = army_cav_scale
                multiply = 0.025
            }
            change_variable = {
                name = army_art_scale
                multiply = 0.01
            }
            capital = {
                create_army_country_in_location = {
                    overlord = scope:target
                    culture = scope:actor.culture
                    religion = scope:actor.religion
                    subject_type = subject_type:vassal
                    ruler = scope:target_character
                    save_scope_as = new_army_country
                }
            }
        }
        scope:new_army_country = {
            add_reform = government_reform:landless_adventurer_reform
            if = {
                limit = {
                    is_member_of_international_organization = international_organization:hre
                }
                remove_from_international_organization = international_organization:hre
            }
            scope:actor.capital = {
                while = {
                    limit = {
                        scope:actor.var:army_inf_scale > 0
                    }
                    scope:actor = {
                        change_variable = {
                            name = army_inf_scale
                            subtract = 1
                        }
                    }
                    create_sub_unit_with_owner = {
                        type = a_pikemen
                        owner = scope:new_army_country
                        origin = scope:actor.capital
                        experience = 50
                    }
                }
                while =  {
                    limit = {
                        scope:actor.var:army_cav_scale > 1
                    }
                    scope:actor = {
                        change_variable = {
                            name = army_cav_scale
                            subtract = 1
                        }
                    }
                    create_sub_unit_with_owner = {
                        type = a_pistoleers
                        owner = scope:new_army_country
                        origin = scope:actor.capital
                        experience = 50
                    }
                }
                while =  {
                    limit = {
                        scope:actor.var:army_art_scale > 1
                    }
                    scope:actor = {
                        change_variable = {
                            name = army_art_scale
                            subtract = 1
                        }
                    }
                    create_sub_unit_with_owner = {
                        type = a_chambered_cannon
                        owner = scope:new_army_country
                        origin = scope:actor.capital
                        experience = 50
                    }
                }
            }
            #after giving them the army we need to recalc the country active status
            force_recalc_country_active_status = scope:new_army_country
            #Give them enough money to sustain their army for some years
            add_gold = { value = scope:actor.monthly_income_total }
            #Add Military Access
            create_relation = {
                first = scope:new_army_country
                second = scope:actor
                type = relation_type:military_access
                months = 12
            }
            #Max Morale
            every_army = {
                every_sub_unit = {
                    add_subunit_morale = {
                        value = subunit_morale
                        divide = subunit_morale_percentage
                        subtract = subunit_morale
                    }
                }
            }

            #move the character into the new country
            scope:target_character = {
                move_country = scope:new_army_country
            }

            random_army = {
                limit = {
                    unit_has_leader = no
                }
                set_as_commander = scope:target_character
            }
            #Copy the possible laws form actor
            scope:actor = {
                every_current_policy = {
                    scope:new_army_country = {
                        add_policy = prev
                    }
                }
            }

            join_war_with = {
                target = scope:target
                war = "scope:target.war_with_country(scope:target_2)"
            }
        }
        scope:actor = {
            trigger_event_non_silently = the_deluge.6
        }
    }

    ai_will_do = {
        value = -30
        if = {
            limit = {
                scope:actor = {
                    is_allied_with = {
                        target = scope:target
                    }
                }
            }
            add = 25
        }
        if = {
            limit = {
                scope:actor = {
                    has_positive_opinion = scope:target
                }
            }
            add = 10
        }
        if = {
            limit = {
                scope:target = {
                    this = c:PLC
                }
                scope:actor = {
                    OR = {
                        this = c:HAB
                        this = c:WAL
                        #this = c:DAN
                    }
                }
            }
            add = 25
        }
        if = {
            limit = {
                scope:target = {
                    this = c:SWE
                }
                scope:actor = {
                    OR = {
                        this = c:BRA
                    }
                }
            }
            add = 20
        }
    }
}
