    show_message = no

    ai_tick_frequency = 12

    automation_tick = never
    automation_tick_frequency = 12

    price = price:claim_mandate_of_heave_prices

    cooldown = {
        type = claim_mandate_of_heaven
        years = 20
    }

    potential = {
        scope:actor = {
            capital = {
                OR = {
                    region = region:east_china_region
                    region = region:north_china_region
                    region = region:south_china_region
                    region = region:west_china_region
                    region = region:manchuria_region
                }
            }
            NOT = { is_leader_of_international_organization = international_organization:middle_kingdom }
        }
    }
    allow = {
        scope:actor = {
            capital = {
                OR = {
                    region = region:east_china_region
                    region = region:north_china_region
                    region = region:south_china_region
                    region = region:west_china_region
                    region = region:manchuria_region
                }
            }
            NOT = { is_leader_of_international_organization = international_organization:middle_kingdom }
            OR = {
                owns = location:dadu
                owns = location:shangyuan
            }
            OR = {
                owns = location:luoyang
                owns = location:xiangyang
            }
            OR = {
                owns = location:changan
                owns = location:chengdu
                owns = location:jiangxia
                owns = location:xuzhou
            }
            num_locations_owned_or_owned_by_subjects_or_below >= 800
        }
        NOT = { is_situation_active = situation:late_ming_crisis }
    }
    effect = {
        scope:actor = {
            change_government_type = government_type:monarchy
            rtr_recreate_middle_kingdom = yes
            #set_country_rank_effect = { rank = country_rank:rank_empire }
            set_country_rank = country_rank:rank_empire
        }
    }

    ai_will_do = { value = 100 }	#They always want to do this
}
