# 明朝危机局势 - 获取统一中国CB的按钮
late_ming_crisis_grant_unification_cb = {
	type = situation
	
	potential = {
		scope:actor = {
			can_see_situation = situation:late_ming_crisis
		}
	}

	allow = {
		scope:actor = {
			capital ?= {
				OR = {
					region = region:west_china_region
					region = region:east_china_region
					region = region:north_china_region
					region = region:south_china_region
					region = region:manchuria_region
				}
			}
		}
	}

	ai_tick = monthly
	ai_tick_frequency = 2
	automation_tick = never
	automation_tick_frequency = 3

	select_trigger = {
		looking_for_a = country
		source_global_list = china_countries_list
		target_flag = target
		name = "choose_target_country"
		column = {
			data = name
		}
		column = {
			data = military_strength
		}
		visible = {
			NOT = { this = scope:actor }
			# 检查目标是否是邻居
			scope:actor = {
				any_neighbor_country = {
					this = root
				}
			}
		}
	}

	effect = {
		if = {
			limit = {
				exists = scope:target
			}
			scope:actor = {
				add_casus_belli = {
					target = scope:target
					type = casus_belli:cb_chinese_unification
				}
			}
		}
	}

	ai_will_do = {
		# 基础分数
		add = 50
		
		# 检查目标是否是邻居
		if = {
			limit = {
				exists = scope:target
				scope:actor = {
					any_neighbor_country = {
						this = scope:target
					}
				}
				# 如果已经拥有对该目标的CB，降低优先级
				scope:actor = {
					NOT = { 
						has_casus_belli_of_type_on = {
							type = casus_belli:cb_chinese_unification
							target = scope:target
						}
					}
				}
			}
			value = 150
		}
		
		# 如果已有CB，降低分数
		if = {
			limit = {
				scope:actor = {
					has_casus_belli_of_type_on = {
						type = casus_belli:cb_chinese_unification
						target = scope:target
					}
				}
			}
			value = -200
		}
	}
}

# 讨伐不臣 - 对附庸获取统一中国CB

	type = situation
	
	potential = {
		scope:actor = {
			can_see_situation = situation:late_ming_crisis
		}
	}

	allow = {
		scope:actor = {
			capital ?= {
				OR = {
					region = region:west_china_region
					region = region:east_china_region
					region = region:north_china_region
					region = region:south_china_region
					region = region:manchuria_region
				}
			}
		}
	}

	ai_tick = monthly
	ai_tick_frequency = 2
	automation_tick = never
	automation_tick_frequency = 3

	select_trigger = {
		looking_for_a = country
		target_flag = target
		name = "choose_target_country"
		column = {
			data = name
		}
		column = {
			data = military_strength
		}
		visible = {
			NOT = { this = scope:actor }
			# 检查目标是否是actor的附庸
			is_subject_or_below_of = scope:actor
			# 使用scripted trigger检查是否在中国区域
			zzz_1644_is_china_region_country = yes
		}
	}

	effect = {
		if = {
			limit = {
				exists = scope:target
				scope:target = {
					is_subject_or_below_of = scope:actor
				}
			}
			# 解除附庸关系
			scope:actor = {
				cancel_subject = scope:target
			}
			# 添加统一中国CB
			scope:actor = {
				add_casus_belli = {
					target = scope:target
					type = casus_belli:cb_chinese_unification
				}
			}
		}
	}

	ai_will_do = {
		# 基础分数
		add = 50
		
		# 检查目标是否是附庸
		if = {
			limit = {
				exists = scope:target
				scope:target = {
					is_subject_or_below_of = scope:actor
				}
				# 如果已经拥有对该目标的CB，降低优先级
				scope:actor = {
					NOT = { 
						has_casus_belli_of_type_on = {
							type = casus_belli:cb_chinese_unification
							target = scope:target
						}
					}
				}
			}
			value = 150
		}
		
		# 如果已有CB，降低分数
		if = {
			limit = {
				scope:actor = {
					has_casus_belli_of_type_on = {
						type = casus_belli:cb_chinese_unification
						target = scope:target
					}
				}
			}
			value = -200
		}
	}
}

# 重建天朝 - 在明末危机局势中重建天朝组织
late_ming_crisis_restore_celestial_empire = {
	type = situation
	
	potential = {
		scope:actor = {
			can_see_situation = situation:late_ming_crisis
		}
	}

	allow = {
		scope:actor = {
			zzz_1644_is_china_region_country = yes
			OR = {
				NOT = { exists = international_organization:middle_kingdom }
				NOT = { is_leader_of_international_organization = international_organization:middle_kingdom }
			}
			# 需要占领所有主要城市以证明统一
			zzz_1644_owns_major_cn_cities = yes
		}
	}

	ai_tick = monthly
	ai_tick_frequency = 2
	automation_tick = never
	automation_tick_frequency = 3

	select_trigger = {
		looking_for_a = situation
		target_flag = recipient
		name = "choose_situation"
		column = {
			data = name
		}
		visible = {
			situation:late_ming_crisis = this
			situation_is_active = yes
		}
	}

	effect = {
		scope:actor = {
			zzz_1644_recreate_middle_kingdom = yes
		}
	}

	ai_will_do = {
		# 基础分数
		add = 100
		
		# 如果已经统一了中国（拥有多个关键城市），提高优先级
		if = {
			limit = {
				scope:actor = {
					owns = location:dadu
					owns = location:chengdu
					owns = location:jingzhao
					owns = location:jiangxia
					owns = location:shangyuan
					stability > 50
				}
			}
			value = 200
		}
		
		# 如果已经是天朝领袖，降低优先级
		if = {
			limit = {
				scope:actor = {
					is_leader_of_international_organization = international_organization:middle_kingdom
				}
			}
			value = -500
		}
	}
}
