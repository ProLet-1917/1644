# 明朝危机局势 - 获取统一中国CB的按钮
late_ming_crisis_grant_unification_cb = {
	type = situation
	
	potential = {
		scope:actor = {
			can_see_situation = situation:late_ming_crisis
		}
	}

	allow = {
		scope:actor = {
			capital ?= {
				OR = {
					region = region:west_china_region
					region = region:east_china_region
					region = region:north_china_region
					region = region:south_china_region
					region = region:manchuria_region
				}
			}
		}
	}

	ai_tick = monthly
	ai_tick_frequency = 3 # check every 6 months
	automation_tick = never
	automation_tick_frequency = 3

	select_trigger = {
		looking_for_a = country
		target_flag = target
		name = "choose_target_country"
		column = {
			data = name
		}
		column = {
			data = military_strength
		}
		visible = {
			exists = capital
			NOT = { this = scope:actor }
			NOT = { is_subject_or_below_of = scope:actor }
			OR = {
				capital.region = region:west_china_region
				capital.region = region:east_china_region
				capital.region = region:north_china_region
				capital.region = region:south_china_region
				capital.region = region:manchuria_region
			}
			any_owned_location = {
				OR = {
					region = region:west_china_region
					region = region:east_china_region
					region = region:north_china_region
					region = region:south_china_region
					region = region:manchuria_region
				}
			}
		}
	}

	effect = {
		if = {
			limit = {
				exists = scope:target
				scope:target = {
					exists = capital
					OR = {
						capital.region = region:west_china_region
						capital.region = region:east_china_region
						capital.region = region:north_china_region
						capital.region = region:south_china_region
						capital.region = region:manchuria_region
					}
					any_owned_location = {
						OR = {
							region = region:west_china_region
							region = region:east_china_region
							region = region:north_china_region
							region = region:south_china_region
							region = region:manchuria_region
						}
					}
				}
			}
			scope:actor = {
				add_casus_belli = {
					target = scope:target
					type = casus_belli:cb_chinese_unification
				}
			}
		}
	}

	ai_will_do = {
		add = 50
		add = {
			if = {
				limit = {
					scope:actor = {
						any_country = {
							limit = {
								exists = capital
								NOT = { this = scope:actor }
								NOT = { is_subject_or_below_of = scope:actor }
								NOT = { has_truce_with = scope:actor }
								OR = {
									capital.region = region:west_china_region
									capital.region = region:east_china_region
									capital.region = region:north_china_region
									capital.region = region:south_china_region
									capital.region = region:manchuria_region
								}
							}
							count >= 1
						}
					}
				}
				value = 100
			}
		}
	}
}
