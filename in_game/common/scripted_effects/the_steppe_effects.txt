start_herd_check_effect = {
	every_country = {
		limit = {
			uses_government_power = horde_unity
		}
		set_variable = {
			name = current_herd
			value = 0
		}
		clamp_variable = {
			name = current_herd
			max = 100000
			min = -100000
		}
		set_variable = {
			name = herd_monthly_change
			value = 0
		}
		set_variable = {
			name = herd_monthly_income
			value = 0
		}
		set_variable = {
			name = herd_natural_decay
			value = 0
		}
		set_variable = {
			name = herd_from_locations
			value = 0
		}
		set_variable = {
			name = herd_decay_from_corruption
			value = 0
		}
	}
	while = {
		count = 12
		every_country = {
			limit = {
				has_variable = current_herd
				uses_government_power = horde_unity
			}
			change_variable = {
				name = herd_monthly_change
				subtract = {
					value = var:herd_monthly_change
				}
			}
			change_variable = {
				name = herd_monthly_income
				subtract = {
					value = var:herd_monthly_income
				}
			}
			change_variable = {
				name = herd_natural_decay
				subtract = {
					value = var:herd_natural_decay
				}
			}
			change_variable = {
				name = herd_from_locations
				subtract = {
					value = var:herd_from_locations
				}
			}
			change_variable = {
				name = herd_decay_from_corruption
				subtract = {
					value = var:herd_decay_from_corruption
				}
			}
			change_variable = {
				name = herd_monthly_income
				add = {
					value = monthly_income_trade_and_tax
					multiply = 0.33
				}
			}
			if = {
				limit = {
					pop_type_population_in_country = {
						pop_type = pop_type:herdsman
						value >= 0
					}
				}
				change_variable = {
					name = herd_from_locations
					add = {
						value = total_control_scaled_population
						multiply = percentage_pop_type_in_country:herdsman
						multiply = 0.5
					}
				}
			}
			change_variable = {
				name = current_herd
				add = {
					value = var:herd_monthly_income
				}
			}
			change_variable = {
				name = current_herd
				add = {
					value = var:herd_from_locations
				}
			}
			if = {
				limit = {
					var:current_herd >= 0
				}
				change_variable = {
					name = herd_natural_decay
					add = {
						value = var:current_herd
						multiply = 0.01
					}
				}
				change_variable = {
					name = herd_decay_from_corruption
					add = {
						value = var:current_herd
						multiply = total_control_scaled_population
						divide = total_population
						multiply = 0.05
					}
				}
			}
			change_variable = {
				name = current_herd
				subtract = {
					value = var:herd_natural_decay
				}
			}
			change_variable = {
				name = current_herd
				subtract = {
					value = var:herd_decay_from_corruption
				}
			}
			change_variable = {
				name = herd_monthly_change
				add = {
					value = var:herd_monthly_income
				}
			}
			change_variable = {
				name = herd_monthly_change
				add = {
					value = var:herd_from_locations
				}
			}
			change_variable = {
				name = herd_monthly_change
				subtract = {
					value = var:herd_natural_decay
				}
			}
			change_variable = {
				name = herd_monthly_change
				subtract = {
					value = var:herd_decay_from_corruption
				}
			}
			set_variable = {
				name = temp_monthly_herd_check
				days = 10
			}
		}
	}
}

strength_horde_start_effect = {
	custom_description = {
		every_country = {
			limit = {
				government_type = government_type:steppe_horde
			}
			add_to_global_variable_list = { name = horde_nations_s_list target = this }

		}
		ordered_in_global_list = {  
			variable = horde_nations_s_list
			limit = { is_subject = no }
			order_by = {
				
				add = {
					value = military_strength
				}
				add = {
					value = monthly_income_trade_and_tax
				}
			} 
			max = 1
			check_range_bounds = no 
			situation:the_steppe = {
				set_variable = {
					name = strongest_strongest_horde_variable
					value = prev
				}
				set_variable = {
					name = strongest_horde_total_score_variable
					value = {
						add = var:strongest_horde_variable.military_strength
						add = var:strongest_horde_variable.monthly_income_trade_and_tax 
					}
				}
			}
			#trigger_event_non_silently =
		}
		ordered_in_global_list = {  
			variable = horde_nations_s_list
			limit = { 
				this != situation:the_steppe.var:strongest_horde_variable 
				is_subject = no
			}
			order_by = {

			variable = horde_nations_s_list

				this != situation:rise_of_the_ottomans.var:strongest_horde_variable 
				this != situation:rise_of_the_ottomans.var:second_strongest_strongest_horde_variable

recalculate_the_strength_horde_monthly_effect  = {
	custom_description = {
		ordered_in_global_list = {  
			variable = horde_nations_s_list
			limit = { is_subject = no }
			order_by = {
			
				add = {
					value = military_strength
				}
				add = {
					value = monthly_income_trade_and_tax
				}
				if = {
					limit = { is_subject = yes }
					multiply = 0.75
				}
			} 
			max = 1
			check_range_bounds = no 
			situation:the_steppe = {
				set_variable = {
					name = strongest_strongest_horde_variable
					value = prev
				}
				set_variable = {
					name = strongest_horde_total_score_variable
					value = {
						add = var:strongest_horde_variable.military_strength
						add = var:strongest_horde_variable.monthly_income_trade_and_tax 
					}
				}
			}
		}	
		if = {
			limit = {
				global_variable_list_size = {
					name = horde_nations_s_list
					value > 1
				}
			}
			ordered_in_global_list = {	#2nd strongest
