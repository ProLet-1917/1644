# 收集所有首都在中国和满洲的国家到全局变量列表
# 每月更新，用于GUI动态显示，按军力排序（降序，军力高的在前）
zzz_1644_collect_china_countries_effect = {
	clear_global_variable_list = china_countries_list
	every_country = {
		limit = {
			exists = capital
			exists = this
			is_subject = no
			OR = {
				capital.region = region:west_china_region
				capital.region = region:east_china_region
				capital.region = region:north_china_region
				capital.region = region:south_china_region
				capital.region = region:manchuria_region
			}
		}
		add_to_global_variable_list = {
			name = china_countries_list
			target = this
		}
	}
	# 按军力排序（降序，军力高的在前）
	ordered_in_global_list = {
		variable = china_countries_list
		order_by = {
			add = {
				value = military_strength
			}
		}
	}
}



# 为单个国家授予对指定目标国家的统一中国CB
# 用于generic action按钮，选择目标国家后使用
zzz_1644_grant_unification_cb_for_single_target = {
	# 为当前国家授予对目标国家的CB
	if = {
		limit = {
			exists = scope:target
			scope:target = {
				exists = capital
				OR = {
					capital.region = region:west_china_region
					capital.region = region:east_china_region
					capital.region = region:north_china_region
					capital.region = region:south_china_region
					capital.region = region:manchuria_region
				}
				any_owned_location = {
					OR = {
						region = region:west_china_region
						region = region:east_china_region
						region = region:north_china_region
						region = region:south_china_region
						region = region:manchuria_region
					}
				}
			}
		}
		root = {
			add_casus_belli = {
				target = scope:target
				type = casus_belli:cb_chinese_unification
			}
		}
	}
}
