# 收集所有首都在中国和满洲的国家到全局变量列表
zzz_1644_collect_china_countries_effect = {
	custom_description = {
		clear_global_variable_list = china_countries_list
		every_country = {
			limit = {
				exists = capital
				exists = this
				is_subject = no
				OR = {
					capital.region = region:west_china_region
					capital.region = region:east_china_region
					capital.region = region:north_china_region
					capital.region = region:south_china_region
					capital.region = region:manchuria_region
				}
			}
			add_to_global_variable_list = {
				name = china_countries_list
				target = this
			}
		}
	}
}

# 为每个首都在中国和满洲的国家添加对其他所有国家的CB
# 使用全局变量列表，遍历列表中的国家
zzz_1644_add_china_cb_effect = {
	custom_description = {
		every_country = {
			limit = {
				has_global_variable_list = china_countries_list
				exists = this
			}
			save_scope_as = source_country
			# 遍历所有国家，为在列表中的其他国家添加CB
			every_country = {
				limit = {
					has_global_variable_list = china_countries_list
					NOT = { this = scope:source_country }
					exists = this
				}
				scope:source_country = {
					add_casus_belli = {
						target = this
						type = casus_belli:cb_chinese_unification
					}
				}
			}
		}
	}
}

# 为单个国家授予对所有中国境内国家的统一中国CB
# 用于generic action按钮
zzz_1644_grant_unification_cb_for_country = {
	custom_description = {
		text = zzz_1644_grant_unification_cb_tt
		# 为当前国家授予对所有符合条件的国家的CB
		every_country = {
			limit = {
				exists = capital
				exists = this
				NOT = { this = root }
				NOT = { is_subject_or_below_of = root }
				NOT = { has_truce_with = root }
				OR = {
					capital.region = region:west_china_region
					capital.region = region:east_china_region
					capital.region = region:north_china_region
					capital.region = region:south_china_region
					capital.region = region:manchuria_region
				}
				any_owned_location = {
					OR = {
						region = region:west_china_region
						region = region:east_china_region
						region = region:north_china_region
						region = region:south_china_region
						region = region:manchuria_region
					}
				}
			}
			root = {
				add_casus_belli = {
					target = this
					type = casus_belli:cb_chinese_unification
				}
			}
		}
	}
}
