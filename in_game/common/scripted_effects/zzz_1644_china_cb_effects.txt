# 收集所有首都在中国和满洲的国家到全局变量列表
# 每月更新，用于GUI动态显示，按人口排序（降序，人口高的在前）
zzz_1644_collect_china_countries_effect = {
	clear_global_variable_list = china_countries_list
	every_country = {
		limit = {
			exists = capital
			exists = this
			is_subject = no
			OR = {
				capital.region = region:west_china_region
				capital.region = region:east_china_region
				capital.region = region:north_china_region
				capital.region = region:south_china_region
				capital.region = region:manchuria_region
			}
		}
		add_to_global_variable_list = {
			name = china_countries_list
			target = this
		}
	}
	# 按人口排序（降序，人口高的在前）
	ordered_in_global_list = {
		variable = china_countries_list
		order_by = {
			add = {
				value = total_population
			}
		}
	}
}

# 为所有收集到的中国国家添加局势 modifier
zzz_1644_add_crisis_participant_modifier_effect = {
	every_in_global_list = {
		variable = china_countries_list
		add_country_modifier = {
			modifier = late_ming_crisis_participant
			years = 10
			mode = add_and_extend
		}
	}
}

# 为单个国家授予对指定目标国家的统一中国CB
# 用于generic action按钮，选择目标国家后使用
zzz_1644_grant_unification_cb_for_single_target = {
	# 为当前国家授予对目标国家的CB
	if = {
		limit = {
			exists = scope:target
			scope:target = {
				exists = capital
				OR = {
					capital.region = region:west_china_region
					capital.region = region:east_china_region
					capital.region = region:north_china_region
					capital.region = region:south_china_region
					capital.region = region:manchuria_region
				}
				any_owned_location = {
					OR = {
						region = region:west_china_region
						region = region:east_china_region
						region = region:north_china_region
						region = region:south_china_region
						region = region:manchuria_region
					}
				}
			}
		}
		root = {
			add_casus_belli = {
				target = scope:target
				type = casus_belli:cb_chinese_unification
			}
		}
	}
}

# 为单个国家授予对所有中国境内国家的统一中国CB
# 保留此效果以备后用
# 使用全局列表以提高性能
zzz_1644_grant_unification_cb_for_country = {
	# 为当前国家授予对所有符合条件的国家的CB
	every_in_global_list = {
		variable = china_countries_list
		limit = {
			any_owned_location = {
				OR = {
					region = region:west_china_region
					region = region:east_china_region
					region = region:north_china_region
					region = region:south_china_region
					region = region:manchuria_region
				}
			}
		}
		root = {
			add_casus_belli = {
				target = this
				type = casus_belli:cb_chinese_unification
			}
		}
	}
}

# 重建天朝：创建天朝组织并设置当前国家为leader
zzz_1644_recreate_middle_kingdom = {
	# 保存当前国家scope
	save_scope_as = country_to_become_leader
	# 如果天朝不存在，创建它
	if = {
		limit = { NOT = { exists = international_organization:middle_kingdom } }
		create_international_organization = {
			type = international_organization_type:middle_kingdom
		}
		international_organization:middle_kingdom = {
			set_variable = { name = num_of_celestial_governors value = 0 }
			set_variable = { name = celestial_authority value = 50 }
		}
	}
	# 将当前国家添加为成员并设置为leader
	international_organization:middle_kingdom = {
		add_country_to_international_organization = scope:country_to_become_leader
		set_leader_country = scope:country_to_become_leader
		change_celestial_authority = { value = celestial_authority_very_strong_bonus }
	}
}
