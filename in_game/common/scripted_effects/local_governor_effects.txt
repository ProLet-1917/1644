set_local_governor_control = {
    scope:target = {
        set_variable = local_governor_working_on_this
        add_location_modifier = {
            modifier = local_governor_control
            years = -1
            size = local_governor_mil_value
        }
    }
}

remove_local_governor_actions = {
    if = {
        limit = {
            has_location_modifier = local_governor_control
        }
        remove_variable = local_governor_working_on_this
        remove_location_modifier = local_governor_control
    }
    if = {
        limit = {
            has_location_modifier = local_governor_culture
        }
        remove_variable = local_governor_working_on_this
        remove_location_modifier = local_governor_culture
    }
    if = {
        limit = {
            has_location_modifier = local_governor_religion
        }
        remove_variable = local_governor_working_on_this
        remove_location_modifier = local_governor_religion
    }
    if = {
        limit = {
            has_location_modifier = local_governor_happiness
        }
        remove_variable = local_governor_working_on_this
        remove_location_modifier = local_governor_happiness
    }
    if = {
        limit = {
            has_location_modifier = local_governor_extra_levies
        }
        remove_variable = local_governor_working_on_this
        remove_location_modifier = local_governor_extra_levies
    }
    if = {
        limit = {
            has_location_modifier = local_governor_production
        }
        remove_variable = local_governor_working_on_this
        remove_location_modifier = local_governor_production
    }
}
