# 根据科技等级给予思潮
# 科技等级对应：
# 等级1 = Age 1 (Traditions) - feudalism, legalism, meritocracy（所有地区相同）
# 等级2 = Age 2 (Renaissance) - renaissance, banking, professional_armies（仅欧洲包含renaissance，其他地区不含）
# 等级3 = Age 3 (Discovery) - new_world, printing_press, pike_and_shot（欧洲包含new_world，但奥斯曼除外；其他地区不含；中国地区额外给予artillery_institution）
# 等级4 = Age 4 (Reformation) - confessionalism, global_trade, artillery_institution（所有地区相同）
give_institutions_by_tech_level = {
	if = {
		limit = { has_variable = start_tech_lv_1644 }
		
		# 等级1：Age 1 思潮（所有地区相同）
		if = {
			limit = { var:start_tech_lv_1644 >= 1 }
			give_traditions_institutions = yes
		}
		
		# 等级2：Age 2 思潮
		if = {
			limit = { var:start_tech_lv_1644 >= 2 }
			# 判断是否为欧洲地区（只有欧洲包含文艺复兴）
			if = {
				limit = { exists = capital capital.continent = continent:europe }
				give_renaissance_institutions = yes
			}
			else = {
				# 其他地区（包括中国）不含文艺复兴
				give_age_2_institutions_without_renaissance = yes
			}
		}
		
		# 等级3：Age 3 思潮
		if = {
			limit = { var:start_tech_lv_1644 >= 3 }
			# 判断是否为欧洲地区且不是奥斯曼（只有欧洲包含新大陆，但奥斯曼除外）
			if = {
				limit = { 
					exists = capital 
					capital.continent = continent:europe
					NOT = { tag = TUR } # 奥斯曼不包含新大陆
				}
				give_discovery_institutions = yes
			}
			else = {
				# 其他地区（包括中国、奥斯曼等）不含新大陆
				give_age_3_institutions_without_new_world = yes
				# 中国地区单独给予攻城炮兵思潮
				if = {
					limit = { zzz_1644_is_china_region_country = yes }
					give_artillery_institution = yes
				}
			}
		}
		
		# 等级4：Age 4 思潮（所有地区相同）
		if = {
			limit = { var:start_tech_lv_1644 >= 4 }
			give_reformation_institutions = yes
			
			# 瑞典特殊处理：4级科技时也给予军事革命思潮
			if = {
				limit = { tag = SWE }
				give_military_revolution_institution = yes
			}
		}
	}
}
