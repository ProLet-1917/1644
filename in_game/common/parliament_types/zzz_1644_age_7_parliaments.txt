# Anno 1644 - Age of Imperialism
# Age 7 parliament types.
#
# Design notes:
#  * Seat share of an estate is represented by global_<estate>_estate_power and
#    <estate>_estate_agenda_impact, i.e. the composition of the chamber directly
#    shifts power towards the represented estates.
#  * republican_parliament / presidential_parliament / imperial_parliament all set
#    allow_law_change_by_stability = no. Laws can then no longer simply be bought
#    with stability and must instead be pushed through parliament issues.

# 共和议会
republican_parliament = {
    type = country

    potential = {
        OR = {
            always = yes
            has_dlc = "0_5_dawn_of_the_new_era"
        }
        OR = {
            modifier:allow_republican_parliament = yes
            has_reform = government_reform:republican_constitution_reform
            has_reform = government_reform:parliamentary_government_reform
        }
    }

    allow = {
        NOT = { has_country_modifier = supreme_authority_modifier }
        government_type = government_type:republic
    }

    modifier = {
        has_a_parliamentary_system = yes
        forbid_multiple_policies_vote = yes
        allow_law_change_by_stability = no

        # Seat share: burghers dominate, peasants are represented, nobles lose out.
        burghers_estate_can_participate_in_parliament = yes
        clergy_estate_can_participate_in_parliament = yes
        peasants_estate_can_participate_in_parliament = yes

        global_burghers_estate_power = 0.20
        global_peasants_estate_power = 0.10
        global_nobles_estate_power = -0.10

        burghers_estate_agenda_impact = 0.20
        peasants_estate_agenda_impact = 0.10
        nobles_estate_agenda_impact = -0.10

        parliament_base_support = 0.10
        parliament_duration_modifier = -0.10
        legislative_efficiency = 0.05

        monthly_towards_liberalism = societal_value_monthly_move
        monthly_towards_free_subjects = societal_value_monthly_move
    }
}

        global_nobles_estate_power = 0.15
        global_burghers_estate_power = 0.15
        global_clergy_estate_power = 0.05

        nobles_estate_agenda_impact = 0.15
        burghers_estate_agenda_impact = 0.15
        clergy_estate_agenda_impact = 0.05
        crown_estate_agenda_impact = 0.10

        parliament_base_support = 0.05
        parliament_request_issue_support_needed = -0.05

        monthly_towards_liberalism = societal_value_monthly_move
        monthly_towards_conciliatory = societal_value_monthly_move
    }
}
