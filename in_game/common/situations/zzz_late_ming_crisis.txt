late_ming_crisis = {
	monthly_spawn_chance = 100
	hint_tag = hint_red_turban_rebellions

	# 1. 触发条件
	can_start = {
		current_year >= 1627
		current_year <= 1650
		exists = c:MNG
		# c:MNG = { owns = location:dadu }
		# OR = {
		# 	current_year = 1644  # 1644年游戏开始时自动启动
		# 	c:MNG = { stability < 50 }
		# 	c:MNG = { current_celestial_authority < 50 } # Assuming this trigger exists for Mandate
		# 	c:MNG = { is_at_war_with = c:QNG }
		# 	c:MNG = { is_at_war_with = c:CSH }
		# }
	}

	# 2. 结束条件
	can_end = {
		OR = {
			current_year > 1690
			
			# # 参与者中有国家同时拥有九个关键城市且稳定度>40
			# # 且局势已运行至少1年（避免开局立即结束）
			# AND = {
			# 	current_year >= 1645
			# 	late_ming_crisis_unification_achieved = yes
			# }
		}
	}

	# 3. 可见性 - 对所有首都在中国和满洲的国家可见
	visible = {
		OR = {
			capital.region = region:west_china_region
			capital.region = region:east_china_region
			capital.region = region:north_china_region
			capital.region = region:south_china_region
			capital.region = region:manchuria_region
			capital.region = region:japan_region
			capital.region = region:korea_region
			capital.region = region:mongolia_region
			capital.region = region:xinjiang_region
		}
	}

	# 4. 启动效果
	on_start = {
		# 解散天朝：移除所有天朝成员
		if = {
			limit = { exists = international_organization:middle_kingdom }
			international_organization:middle_kingdom = {
				# 先移除leader（如果有）
				if = {
					limit = { international_organization_has_leader = yes }
					update_leader_country = yes
				}
				# 移除所有成员
				every_international_organization_member = {
					international_organization:middle_kingdom = {
						remove_country_from_international_organization = prev
					}
				}
			}
		}
		
		# 给所有首都在中国和满洲的国家互相添加"统一中国"CB
		# 使用全局变量列表方法，分两步执行
		zzz_1644_collect_china_countries_effect = yes
		zzz_1644_add_china_cb_effect = yes
		
		# 为初始参与国添加局势 modifier（使用已收集的国家列表）
		zzz_1644_add_crisis_participant_modifier_effect = yes


		#为明、顺、清设置独特变量值
		c:MNG = {
			if = {
				limit = {
					has_variable = mng_move_to_nanjing
				}
				change_variable = {
					name = chi_domination
					add = 20
				}
				change_variable = {
					name = chi_mandate
					add = 30
				}
			}
			else = {
				change_variable = {
					name = chi_domination
					subtract = 20
				}
				change_variable = {
					name = chi_mandate
					add = 40
				}
			}
		}
		c:CSH = {
			if = {
				limit = {
					has_variable = csh_attack_beijing
				}
				change_variable = {
					name = chi_domination
					add = 75
				}
				change_variable = {
					name = chi_mandate
					add = 45
				}
			}
			else = {
				change_variable = {
					name = chi_domination
					add = 50
				}
				change_variable = {
					name = chi_mandate
					add = 60
				}
			}
		}
		c:QNG = {
			change_variable = {
				name = chi_domination
				add = 75
			}
			change_variable = {
				name = chi_mandate
				subtract = 20
			}
		}

	}

	# 5. 月度逻辑
	on_monthly = {
		# 每月更新中国国家列表（用于GUI显示）
		zzz_1644_collect_china_countries_effect = yes
		
		# 为参与国添加局势 modifier（使用已收集的国家列表，只添加给还没有 modifier 的国家）
		every_in_global_list = {
			variable = china_countries_list
			limit = {
				NOT = { has_country_modifier = late_ming_crisis_participant }
			}
			add_country_modifier = {
				modifier = late_ming_crisis_participant
				years = 10
				mode = add_and_extend
			}
		}
		
	}

	# 6. 结束效果
	on_end = {
		# 移除所有参与国的局势 modifier
		every_country = {
			limit = {
				has_country_modifier = late_ming_crisis_participant
			}
			remove_country_modifier = late_ming_crisis_participant
		}
	}

	# 7. 地图颜色显示
	map_color = {
		if = {
			limit = {
				has_owner = yes
				owner ?= { tag = MNG }
			}
			value = orange  # 大明 -
		}
		else_if = {
			limit = {
				has_owner = yes
				owner ?= { tag = CSH }
			}
			value = blue  # 大顺 - 
		}
		else_if = {
			limit = {
				has_owner = yes
				owner ?= { tag = QNG }
			}
			value = yellow  # 大清 - 
		}
		else_if = {
			limit = {
				has_owner = yes
				owner ?= {
					has_country_modifier = late_ming_crisis_participant
				}
			}
			value = green  # 其他参与国 - 
		}
		else = {
			value = define:NMapColors|DEFAULT_COLOR
		}
	}

	legend_key = {
		desc = "MNG"
		color = yellow
		require_color_on_map = yes
	}
	legend_key = {
		desc = "CSH"
		color = red
		require_color_on_map = yes
	}
	legend_key = {
		desc = "QNG"
		color = blue
		require_color_on_map = yes
	}
	legend_key = {
		desc = "late_ming_crisis_participant"
		color = green
		require_color_on_map = yes
	}

}
