late_ming_crisis = {
	monthly_spawn_chance = 100
	hint_tag = hint_red_turban_rebellions

	# 1. 触发条件
	can_start = {
		current_year >= 1627
		current_year <= 1650
		exists = c:MNG
		NOT = { c:MNG = { owns = location:dadu } }
	}

	# 2. 结束条件
	can_end = {
		end_reason = {
			trigger = {
				OR = {
					current_year >= 1685

					# 重建天朝：如果有国家成为了天朝组织的领袖，局势结束
					AND = {
						current_year >= 1645
						exists = international_organization:middle_kingdom
						international_organization:middle_kingdom = {
							international_organization_has_leader = yes
						}
					}
				}
			}
			desc = "new_dynasty_formed"
		}
	}

	# 3. 可见性 - 对所有首都在中国和满洲的国家可见
	visible = {
		OR = {
			capital.region = region:west_china_region
			capital.region = region:east_china_region
			capital.region = region:north_china_region
			capital.region = region:south_china_region
			capital.region = region:manchuria_region
			capital.region = region:mongolia_region
			capital.region = region:xinjiang_region
		}
	}

	# 4. 启动效果
	on_start = {

		# 解散天朝：移除所有天朝成员
		if = {
			limit = { exists = international_organization:middle_kingdom }
			international_organization:middle_kingdom = {
				# 先移除leader（如果有）
				if = {
					limit = { international_organization_has_leader = yes }
					update_leader_country = yes
				}
				# 移除所有成员
				every_international_organization_member = {
					international_organization:middle_kingdom = {
						remove_country_from_international_organization = prev
					}
				}
			}
		}
		
		# 给所有首都在中国和满洲的国家互相添加"统一中国"CB
		# 使用全局变量列表方法，分两步执行
		collect_china_countries_effect = yes
		zzz_1644_add_china_cb_effect = yes
		
		# 为初始参与国添加局势 modifier（使用已收集的国家列表）
		zzz_1644_add_crisis_participant_modifier_effect = yes

		#为所有参与国添加domination和mandate
		late_ming_crisis_init_variable = yes


		#为明、顺、清设置独特变量值
		c:MNG = {
			if = {
				limit = {
					has_variable = mng_move_to_nanjing
				}
				change_variable = {
					name = chi_domination
					add = 20
				}
				change_variable = {
					name = chi_mandate
					add = 30
				}
			}
			else = {
				change_variable = {
					name = chi_domination
					subtract = 20
				}
				change_variable = {
					name = chi_mandate
					add = 40
				}
			}
		}
		c:CSH = {
			if = {
				limit = {
					has_variable = csh_attack_beijing
				}
				change_variable = {
					name = chi_domination
					add = 75
				}
				change_variable = {
					name = chi_mandate
					add = 45
				}
			}
			else = {
				change_variable = {
					name = chi_domination
					add = 50
				}
				change_variable = {
					name = chi_mandate
					add = 60
				}
			}
		}
		c:QNG = {
			change_variable = {
				name = chi_domination
				add = 75
			}
			change_variable = {
				name = chi_mandate
				subtract = 20
			}
		}

		# 初始化明末局势死亡总人数计数器（局势变量）
		set_variable = {
			name = late_ming_crisis_total_casualties
			value = 0
		}
		# 注意：我方击杀和阵亡计数器是国家变量，不需要在局势中初始化

	}

	# 5. 月度逻辑
	on_monthly = {
		# # 每月更新中国国家列表，已经在别处更新
		collect_china_countries_effect = yes
		#为所有参与国添加domination和mandate
		late_ming_crisis_init_variable = yes
		
		# 为参与国添加局势 modifier（使用已收集的国家列表，只添加给还没有 modifier 的国家）
		every_in_global_list = {
			variable = china_countries_list
			limit = {
				NOT = { has_country_modifier = late_ming_crisis_participant }
			}
			add_country_modifier = {
				modifier = late_ming_crisis_participant
				years = 5
				mode = replace
			}
		}
		
		#大同叛乱事件
		if = {
			limit = {
				c:CJX ?= {
					NOR = {
						has_variable = cjx_rebellion_first
						has_variable = cjx_refus_rebellion_first
					}
				}
			}
			if = {
				limit = {
					c:QNG ?= { has_variable = chi_domination }
					c:MNG ?= { has_variable = chi_domination }
					c:CSH ?= { has_variable = chi_domination }
					OR = {
						c:QNG.var:chi_domination > c:CSH.var:chi_domination			
				 		c:MNG.var:chi_domination > c:CSH.var:chi_domination	
					}		
				}
				random = {
					chance = 10
					modifier = {
						c:CSH.var:chi_domination < 70
						multiply = 4
					}
					modifier = {
						c:CSH.var:chi_domination < 60
						multiply = 2
					}
					modifier = {
						c:CSH.var:chi_domination < 50
						multiply = 2
					}
					modifier = {
						location:xuande.owner = c:QNG#宣府
						multiply = 4
					}
					modifier = {
						location:yangqu.owner = c:QNG#太原
						multiply = 4
					}
					modifier = {
						location:ninghua.owner = c:QNG#宁武
						multiply = 2
					}
					modifier = {
						location:daizhou.owner = c:QNG#代州
						multiply = 2
					}
					modifier = {
						location:dadu.owner = c:QNG#北京
						multiply = 4
					}
					modifier = {
						location:jingzhao.owner = c:QNG#西京
						multiply = 4
					}
					modifier = {
						c:CSH.var:chi_domination > 75
						divide = 2
					}
					c:CJX = {
						trigger_event_silently = {
							id = late_ming_crisis.1
							days = { 3 10 }
						}
					}
				}
			}
		}

		#宣府叛乱事件
		if = {
			limit = {
				NOT = { c:CSH ?= { has_variable = csh_lost_xuanfu } }
			}
			if = {
				limit = {
					c:QNG ?= { has_variable = chi_domination }
					c:MNG ?= { has_variable = chi_domination }
					c:CSH ?= { has_variable = chi_domination }
					c:QNG.var:chi_domination > c:CSH.var:chi_domination
				}
				random = {
					chance = 10
					modifier = {
						c:CSH.var:chi_domination < 70
						multiply = 4
					}
					modifier = {
						c:CSH.var:chi_domination < 60
						multiply = 2
					}
					modifier = {
						c:CSH.var:chi_domination < 50
						multiply = 2
					}
					modifier = {
						OR = {
							location:datong_datong.owner = c:QNG
							c:CJX ?= { is_subject_of = c:QNG }
						}
						multiply = 2
					}
					modifier = {
						location:dadu.owner = c:QNG#北京
						multiply = 2
					}
					modifier = {
						c:CSH.var:chi_domination > 75
						divide = 2
					}
					c:CSH = {
						trigger_event_silently = {
							id = late_ming_crisis.3
							days = { 1 3 }
						}
					}
				}
			}
		}

		#河北山东叛乱事件
		if = {
			limit = { NOT = { has_global_variable = csh_rebellion_of_hebe_shandong } }
			if = {
				limit = {
					c:QNG ?= { has_variable = chi_domination }
					c:MNG ?= { has_variable = chi_domination }
					c:CSH ?= { has_variable = chi_domination }
					c:CSH.var:chi_mandate < c:MNG.var:chi_mandate
				}
				random = {
					chance = 25
					modifier = {
						c:CSH.var:chi_mandate < 50
						multiply = 2
					}
					modifier = {
						c:CSH.var:chi_mandate < 40
						multiply = 2
					}
					modifier = {
						location:dadu.owner != c:CSH#北京
						multiply = 2
					}
					modifier = {
						c:QNG.var:chi_domination > c:CSH.var:chi_domination	
						multiply = 2
					}
					modifier = {
						c:CSH.var:chi_domination > 75
						divide = 2
					}
					c:CSH = {
						trigger_event_silently = {
							id = late_ming_crisis.4
							days = { 1 3 }
						}
					}
				}
			}
		}

		#放弃北京
		if = {
			limit = { NOT = { c:CSH = { has_variable = csh_back_to_shaanxi } } }
			if = {
				limit = {
					c:QNG ?= { has_variable = chi_domination }
					c:MNG ?= { has_variable = chi_domination }
					c:CSH ?= { 
						has_variable = chi_domination 
						has_variable = csh_attack_beijing
					}
					OR = {
						c:QNG.var:chi_domination > c:CSH.var:chi_domination			
				 		c:MNG.var:chi_domination > c:CSH.var:chi_domination	
					}
				}
				random = {
					chance = 5
					modifier = {
						c:CSH = { has_variable = csh_lost_xuanfu }
						multiply = 4
					}
					modifier = {
						OR = {
							location:funing.owner = c:QNG
							c:WSG ?= { is_subject_of = c:QNG }
						}
						multiply = 4
					}
					modifier = {
						has_global_variable = csh_rebellion_of_hebe_shandong
						multiply = 4
					}
					modifier = {
						c:CJX ?= { is_subject_of = c:QNG }
						multiply = 2
					}
					modifier = {
						c:CSH.var:chi_domination < 60
						multiply = 2
					}
					modifier = {
						c:CSH.var:chi_domination > 75
						divide = 2
					}
					modifier = {
						c:CSH.var:chi_mandate < 25
						multiply = 2
					}
					c:CSH = {
						trigger_event_silently = {
							id = late_ming_crisis.5
							days = { 1 3 }
						}
					}
				}
			}
		}

		#济王投降
		if = {
			limit = {
				NOT = { has_global_variable = cji_try_surrender_to_qng }
			}
			if ={
				limit = {
					c:QNG ?= { has_variable = chi_domination }
					c:MNG ?= { has_variable = chi_domination }
					c:CSH ?= { has_variable = chi_domination }
					c:QNG.var:chi_domination > c:MNG.var:chi_domination
				}
				random = {
					chance = 10
					modifier = {
						location:dadu.owner = c:QNG
						multiply = 2
					}
					modifier = {
						c:QNG.var:chi_domination > 80
						multiply = 2
					}
					modifier = {
						c:CJI = {
							is_at_war_with = c:QNG
						}
						multiply = 2
					}
					c:CJI = {
						trigger_event_silently = {
							id = late_ming_crisis.6
							days = { 1 3 }
						}
					}
				}
			}
		}

	}

	# 6. 结束效果
	on_ended = {
		# 移除所有参与国的局势 modifier
		every_country = {
			limit = {
				has_country_modifier = late_ming_crisis_participant
			}
			remove_country_modifier = late_ming_crisis_participant
		}
	}

	# 7. 地图颜色显示
	map_color = {
		if = {
			limit = {
				has_owner = yes
				owner ?= { tag = MNG }
			}
			value = orange  # 大明 -
		}
		else_if = {
			limit = {
				has_owner = yes
				owner ?= { tag = CSH }
			}
			value = blue  # 大顺 - 
		}
		else_if = {
			limit = {
				has_owner = yes
				owner ?= { tag = QNG }
			}
			value = yellow  # 大清 - 
		}
		else_if = {
			limit = {
				has_owner = yes
				owner ?= {
					has_country_modifier = late_ming_crisis_participant
				}
			}
			value = green  # 其他参与国 - 
		}
		else = {
			value = define:NMapColors|DEFAULT_COLOR
		}
	}

	legend_key = {
		desc = "MNG"
		color = yellow
		require_color_on_map = yes
	}
	legend_key = {
		desc = "CSH"
		color = red
		require_color_on_map = yes
	}
	legend_key = {
		desc = "QNG"
		color = blue
		require_color_on_map = yes
	}
	legend_key = {
		desc = "late_ming_crisis_participant"
		color = green
		require_color_on_map = yes
	}

}
