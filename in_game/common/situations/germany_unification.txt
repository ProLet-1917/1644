germany_unification = {

    monthly_spawn_chance = monthly_spawn_chance_unique

    can_start = {
        OR = {
            always = yes
            has_dlc = "0_5_dawn_of_the_new_era"
        }
        region:north_german_region = {
            any_location_in_region = {
                percent >= 0.75
                owner = {
                    culture =  { has_culture_group = culture_group:german_group }
                    has_advance = nation_state_advance
                }
                #has_institution = institution:ideology
            }
        }
        region:south_german_region = {
            any_location_in_region = {
                percent >= 0.75
                #is_capital = yes
                owner = {
                    culture =  { has_culture_group = culture_group:german_group }
                    has_advance = nation_state_advance
                }
            }
        }
    }

    outcome = {
    	name = "germany_unification_nation_forged"
    	trigger = {
        OR = {
            OR = {
                country_exists = c:GER
                country_exists = c:HRE
            }
            AND = {
                region:north_german_region = {
                    any_location_in_region = {
                        percent <= 0.2
                        owner = {
                            culture = { has_culture_group = culture_group:german_group }
                        }
                    }
                }
                region:south_german_region = {
                    any_location_in_region = {
                        percent <= 0.2
                        owner = {
                            culture = { has_culture_group = culture_group:german_group }
                        }
                    }
                }
            }
        }
    	}
    }

    visible = {
        capital.continent = continent:europe
    }

    on_start = {
    	add_timeline_note = {
    		key = timeline_note_germany_unification
    		show_in_pop_graph = yes
    		unique = yes
    	}
        every_country = {
            limit = {
                culture = { has_culture_group = culture_group:german_group }
                is_subject = no
                OR = {
                    is_regional_power = yes
                    is_great_power = yes
                    country_rank = country_rank:rank_kingdom
                }
            }
            save_scope_as = target
            add_to_global_variable_list = { name = potential_unifier_list target = this }
            set_variable = potential_unifier
            create_international_organization = {
                type = international_organization_type:germany_unification_union
                add_country_to_international_organization = scope:target
                set_leader_country = scope:target
            }
            if = {
                limit = {
                    has_variable = german_country
                }
                remove_variable = german_country
            }
        }
        every_country = {
            limit = {
                culture = { has_culture_group = culture_group:german_group }
                NOR = {
                    is_regional_power = yes
                    is_great_power = yes
                    country_rank = country_rank:rank_kingdom
                }
                NOT = {
                    has_variable = german_country
                }
            }
            add_to_global_variable_list = { name = german_country_list target = this }
            set_variable = german_country
        }
        every_country = {
            limit = {
                capital.continent = continent:europe
            }
            trigger_event_non_silently = germany_unification.1000
        }
    }

    on_monthly = {
        clear_global_variable_list = german_country_list
        clear_global_variable_list = potential_unifier_list
        ordered_country = {
            limit = {
                culture = { has_culture_group = culture_group:german_group }
                is_subject = no
                OR = {
                    is_regional_power = yes
                    is_great_power = yes
                    country_rank = country_rank:rank_kingdom
                }
                NOT = {
                    has_variable = remove_from_unifier_list
                }
                has_variable = german_country
                NOT = { has_variable = potential_unifier }
                NOT = {
                    is_leader_of_international_organization_of_type = { type = germany_unification_union }
                }
            }
            order_by = total_population
            max = 1
            save_scope_as = target
            remove_variable = german_country
            add_to_global_variable_list = { name = potential_unifier_list target = this }
            set_variable = potential_unifier
            create_international_organization = {
                type = international_organization_type:germany_unification_union
                add_country_to_international_organization = scope:target
                set_leader_country = scope:target
            }
        }
        every_country = {
            limit = {
                has_variable = german_country
            }
            add_to_global_variable_list = { name = german_country_list target = this }
        }
        every_country = {
            limit = {
                has_variable = potential_unifier
            }
            add_to_global_variable_list = { name = potential_unifier_list target = this }
        }

        # 普奥争雄：两名以上统一者，且其中之一为奥地利时
        if = {
            limit = {
                NOT = { has_global_variable = german_brothers_war_fired }
                global_variable_list_size = {
                    name = potential_unifier_list
                    value >= 2
                }
                OR = {
                    AND = {
                        country_exists = c:HAB
                        c:HAB = { has_variable = potential_unifier }
                    }
                    AND = {
                        country_exists = c:AUS
                        c:AUS = { has_variable = potential_unifier }
                    }
                }
            }
            random = {
                chance = 6
                set_global_variable = german_brothers_war_fired
                ordered_country = {
                    limit = {
                        has_variable = potential_unifier
                        NOT = { tag = HAB }
                        NOT = { tag = AUS }
                    }
                    order_by = total_population
                    max = 1
                    trigger_event_silently = {
                        id = germany_unification.1200
                        days = { 3 10 }
                    }
                }
            }
        }
    }

    on_ended = {
    	add_timeline_note = {
    		key = timeline_note_germany_unification_end
    		show_in_pop_graph = yes
    		unique = yes
    	}
        every_country = {
            limit = {
                OR = {
                    has_variable = potential_unifier
                    has_variable = german_country
                }
            }
            remove_variable = potential_unifier
            remove_variable = german_country
        }
        if = {
            limit = { has_global_variable = german_brothers_war_fired }
            remove_global_variable = german_brothers_war_fired
        }
    }

    map_color = {
        if = {
            limit = {
                owner = {
                    has_variable = potential_unifier
                }
            }
            value = rgb { 215 102 112 }
        }
    }

    secondary_map_color = {
        if = {
            limit = {
                owner.culture = { has_culture_group = culture_group:german_group }
            }
            value = rgb { 75 110 182 }
        }
    }
}
