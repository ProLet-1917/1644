peace_talk_in_westphalia = { #1648 May.-Oct.

	monthly_spawn_chance = monthly_spawn_chance_unique

	can_start = {
		exists = international_organization:hre
		situation:thirty_years_war = {
			situation_is_active = no
		}
		has_global_variable = start_peace_talk_in_westphalia
		NOT = {
			has_game_rule = treaty_of_westphalia_historical
		}
	}

	can_end = {
		end_reason = {
			trigger = {
				var:peace_talk_in_westphalia_phase = 5
			}
			desc = "congress_finished"
		}

	}

	visible = {
		OR = {
			is_member_of_international_organization = international_organization:catholic_league
			is_member_of_international_organization = international_organization:protestant_union
		}
	}

	on_start = {
		set_variable = {
			name = peace_talk_in_westphalia_phase_remaining_month
			value = 5
		}
		set_variable = {
			name = peace_talk_in_westphalia_phase
			value = 1
		}
		set_global_variable = {
			name = now_talking_about_north_europe_problem
		}
		set_global_variable = {
			name = losing_side_benifit
			value = 0
		}
		count_allowed_treaty_effect = yes
	}

	on_monthly = {
		change_variable = {
 			name = peace_talk_in_westphalia_phase_remaining_month
  			subtract = 1
		}
		if = {
			limit = {
				var:peace_talk_in_westphalia_phase_remaining_month = 0
			}
			change_variable = {
 				name = peace_talk_in_westphalia_phase
  				add = 1
			}
			change_variable = {
 				name = peace_talk_in_westphalia_phase_remaining_month
  				add = 5
			}
			remove_global_variable = now_talking_about_north_europe_problem
			remove_global_variable = now_talking_about_lowland_problem
			remove_global_variable = now_talking_about_wittersbach_problem
			remove_global_variable = now_talking_about_france_austria_problem
			remove_global_variable = now_talking_about_france_spain_problem
			count_allowed_treaty_effect = yes
		}
		if = {
			limit = {
				var:peace_talk_in_westphalia_phase = 2
			}
			set_global_variable = {
				name = now_talking_about_lowland_problem
			}
		}
		if = {
			limit = {
				var:peace_talk_in_westphalia_phase = 3
			}
			set_global_variable = {
				name = now_talking_about_wittersbach_problem
			}
		}
		if = {
			limit = {
				var:peace_talk_in_westphalia_phase = 4
			}
			set_global_variable = {
				name = now_talking_about_france_austria_problem
			}
		}
		if = {
			limit = {
				var:peace_talk_in_westphalia_phase = 5
			}
			set_global_variable = {
				name = now_talking_about_france_spain_problem
			}
		}
	}

	on_ended = {
		every_country = {
			limit = {
				has_country_modifier = ai_in_religion_war
			}
			remove_country_modifier = ai_in_religion_war
		}
		international_organization:catholic_league.leader_country = {
			trigger_event_silently = thirty_years_war_events.1000
		}
	}

	map_color = {
		if = {
			limit = {
				has_global_variable = catholic_winning
			}
			if = {
				limit = {
					is_member_of_international_organization = international_organization:catholic_league
				}
				value = rgb { 155 99 156 }
			}
			else_if = {
				limit = {
					is_member_of_international_organization = international_organization:protestant_union
				}
				value = rgb { 102 195 223 }
			}
		}
		else_if = {
			limit = {
				has_global_variable = protestant_winning
			}
			if = {
				limit = {
					is_member_of_international_organization = international_organization:catholic_league
				}
				value = rgb { 102 195 223 }
			}
			else_if = {
				limit = {
					is_member_of_international_organization = international_organization:protestant_union
				}
				value = rgb { 155 99 156 }
			}
		}
	}
}
