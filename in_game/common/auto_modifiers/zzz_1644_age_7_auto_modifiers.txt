# Anno 1644 - Age of Imperialism
# Automatic modifiers: whenever a country has one of the new Age 7 parliaments,
# laws can no longer be paid for with stability. They have to be pushed through
# the chamber as parliament issues instead.

republican_parliament_law_vote_required_auto = {
    category = country
    type = country

    potential_trigger = {
        OR = {
            parliament_type = parliament_type:republican_parliament
            parliament_type = parliament_type:presidential_parliament
            parliament_type = parliament_type:imperial_parliament
            parliament_type = parliament_type:mass_suffrage_parliament
        }
    }

    allow_law_change_by_stability = no
    change_policy_cost_modifier = 5.0
    set_policy_cost_modifier = 5.0
}
