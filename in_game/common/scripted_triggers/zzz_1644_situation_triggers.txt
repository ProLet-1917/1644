# 明朝危机局势触发器

# 检查国家是否在中国区域（用于简化重复的区域检查）
zzz_1644_is_china_region_country = {
	exists = capital
	capital ?= {
		OR = {
			region = region:west_china_region
			region = region:east_china_region
			region = region:north_china_region
			region = region:south_china_region
			region = region:manchuria_region
		}
	}
	any_owned_location = {
		OR = {
			region = region:west_china_region
			region = region:east_china_region
			region = region:north_china_region
			region = region:south_china_region
			region = region:manchuria_region
		}
	}
}

# 检查单个国家是否占领了所有主要城市（拥有所有关键城市且稳定度>20）
zzz_1644_owns_major_cn_cities = {
	exists = capital
	is_subject = no
	stability > 20
	# 同时拥有所有关键城市
	owns = location:dadu
	owns = location:chengdu
	owns = location:jingzhao
	owns = location:jiangxia
	owns = location:shangyuan
	owns = location:hangzhou
	owns = location:changsha
	owns = location:pingjin
	owns = location:kaifeng
	owns = location:nanchang
	owns = location:baxian
}

# 检查是否有参与国家满足统一条件（拥有所有关键城市且稳定度>40）
late_ming_crisis_unification_achieved = {
	# 使用参与国列表 china_countries_list 进行检查
	any_in_global_list = {
		variable = china_countries_list
		zzz_1644_owns_major_cn_cities = yes
	}
}

is_chinese_country = {
	AND = {
		OR = {
			capital.region = region:west_china_region
			capital.region = region:east_china_region
			capital.region = region:north_china_region
			capital.region = region:south_china_region
			capital.region = region:manchuria_region
		}
		OR = {
			has_primary_or_accepted_or_tolerated_culture = culture:han_culture
		}
		is_rebel_country = no
	}
}
