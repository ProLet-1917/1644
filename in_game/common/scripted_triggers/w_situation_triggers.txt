
ai_country_should_colonize = {	#Root should be the country
    OR = {
        country_tax_base > 100
        yearly_gold > 0
        tag = SWE
        tag = POR
        tag = KUR
        tag = KNI
        tag = NED
        is_subject_type = colonial_nation
        is_subject_type = charted_company
    }


    OR = {
        trigger_if = {
            limit = {
                has_game_rule = all_countries_colonize
            }
            always = yes
        }
        trigger_else_if = {
            limit = {
                has_game_rule = western_colonisation_only
            }
            this = c:RUS
            this != c:PAP
            not = { has_preference_tag = "warfare" }
            OR = {
                capital.continent = continent:europe
                and = {
                    is_subject = yes
                    overlord = {
                        capital.continent = continent:europe
                    }
                }
            }
        }
        #trigger_else_if = {
        #    limit = {
        #        has_game_rule = all_countries_colonize_except_special
        #    }
        #    this != c:PAP
        #}
    }

    OR = {
        is_subject = no
        is_subject_type = colonial_nation
        is_subject_type = charted_company
    }
}
