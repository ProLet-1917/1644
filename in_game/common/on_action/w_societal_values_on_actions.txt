on_universality_vs_exclusivity_monthly_change = {
    trigger = {
        has_societal_value = societal_value_type:universality_vs_exclusivity
        NOT = {
            has_variable = already_monthly_change_universality_vs_exclusivity
        }
    }
    effect = {
        change_societal_value = {
            type = universality_vs_exclusivity
            value = {
                add = modifier:monthly_towards_exclusivity
                subtract = modifier:monthly_towards_universality
            }
        }
        set_variable = {
            name = already_monthly_change_universality_vs_exclusivity
            days = 10
        }
    }
}
