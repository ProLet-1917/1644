
on_battle_won = {
	effect = {
		#检测是否属于局势参与方
		scope:actor.owner = {
			if = {
				limit = {
					has_variable = chinese_country
					#is_subject = no  # 可选：仅对独立国家生效
				}
				#增加权威
				change_variable = {
					name = chi_domination
					add = scope:war_score
				}
			}
		}
		# 统计明末局势死亡总人数
		# 警告：scope:killed_land_units 和 scope:lost_land_units 动态指向战斗界面左侧的国家（攻击方），不是固定的胜利方
		# 当失败方在左侧时，这些值会指向失败方数据，导致bug
		# 解决方案：通过判断scope:actor是否是攻击方来确定胜利方是否在左侧
		
		# 在scope:actor作用域内读取左侧（攻击方）的数据
		# 两个作用域读取的值相同，都指向左侧（攻击方）
		scope:actor = {
			owner = {
				set_variable = {
					name = left_side_killed
					value = scope:killed_land_units
				}
				set_variable = {
					name = left_side_lost
					value = scope:lost_land_units
				}
			}
		}
		
		# 判断scope:actor是否是攻击方
		scope:actor = {
			if = {
				limit = {
					unit_location = {
						has_combat = yes
						combat ?= {
							combat_attacker ?= {
								any_unit = {
									owner = scope:actor.owner
								}
							}
						}
					}
				}
				# scope:actor是攻击方，胜利方在左侧，使用左侧的值（当前逻辑正确）
				owner = {
					set_variable = {
						name = left_winner_killed
						value = var:left_side_killed
					}
					set_variable = {
						name = left_winner_lost
						value = var:left_side_lost
					}
				}
			}
			else = {
				# scope:actor不是攻击方，失败方在左侧
				# 胜利方的击杀数 = 失败方的损失数 = left_side_lost
				# 胜利方的损失数 = 失败方的击杀数 = left_side_killed
				owner = {
					set_variable = {
						name = right_winner_killed
						value = var:left_side_lost
					}
					set_variable = {
						name = right_winner_lost
						value = var:left_side_killed
					}
				}
			}
		}
		
		# 标记actor和target的身份（用于debug）
		scope:actor.owner = {
			remove_variable = debug_is_target
			set_variable = {
				name = debug_is_actor
				value = yes
			} 
		}
		scope:target.owner = {
			remove_variable = debug_is_actor
			set_variable = {
				name = debug_is_target
				value = yes
			}
		}
	}
}


on_battle_lost = {
	effect = {

		#检测是否属于局势参与方
		scope:actor.owner = {
			if = {
				limit = {
					has_variable = chinese_country
					#is_subject = no  # 可选：仅对独立国家生效
				}
				#削减权威
				change_variable = {
					name = chi_domination
					subtract = scope:war_score
				}
			}
		}

		
	}
}

on_great_battle_lost = {
	effect = {
		if = {
			limit = {
				NOT = { has_global_variable = battle_of_shanhaiguan }
				scope:actor.owner = c:CSH
				scope:target.owner = c:QNG
				scope:actor.unit_location = location:funing
			}
		}
		c:CSH = { trigger_event_silently = csh_1644_unification.6 }
	}
}
