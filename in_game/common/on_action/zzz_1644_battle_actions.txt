
on_battle_won = {
	effect = {
		#检测是否属于局势参与方
		scope:actor.owner = {
			if = {
				limit = {
					has_variable = chinese_country
					#is_subject = no  # 可选：仅对独立国家生效
				}
				#增加权威
				change_variable = {
					name = chi_domination
					add = scope:war_score
				}
			}
		}
		# 统计明末局势死亡总人数
		# 使用 killed_land_units 来计算总伤亡
		# 在 on_battle_won 中：scope:killed_land_units = 获胜方杀死的敌军数（即失败方的损失）
		# 在 on_battle_lost 中：scope:killed_land_units = 失败方杀死的敌军数（即获胜方的损失）
		if = {
			limit = {
				situation:late_ming_crisis = {
					situation_is_active = yes
				}
			}
			# 统计双方都是参与方的情况
			if = {
				limit = {
					AND = {
						scope:actor.owner = {
							has_variable = chinese_country
						}
						scope:target.owner = {
							has_variable = chinese_country
						}
					}
				}
				# 统计总人数（只用 killed_land_units）
				# 获胜方杀死的敌军数（失败方的损失）
				zzz_1644_increase_total_casualties = {
					variable = scope:killed_land_units
				}
				zzz_1644_increase_total_casualties = {
					variable = scope:lost_land_units
				}
			}
		}

on_battle_lost = {
	effect = {
		#检测是否属于局势参与方
		scope:actor.owner = {
			if = {
				limit = {
					has_variable = chinese_country
					#is_subject = no  # 可选：仅对独立国家生效
				}
				#削减权威
				change_variable = {
					name = chi_domination
					subtract = scope:war_score
				}
			}
		}
		
		# 统计明末局势死亡总人数（失败方部分）
		# 在 on_battle_lost 中：scope:killed_land_units = 失败方杀死的敌军数（即获胜方的损失）
		if = {
			limit = {
				situation:late_ming_crisis = {
					situation_is_active = yes
				}
			}
			# 统计双方都是参与方的情况
			if = {
				limit = {
					AND = {
						scope:actor.owner = {
							has_variable = chinese_country
						}
						scope:target.owner = {
							has_variable = chinese_country
						}
					}
				}
			}
		}

		
	}
}

on_great_battle_lost = {
	effect = {
		if = {
			limit = {
				NOT = { has_global_variable = battle_of_shanhaiguan }
				scope:actor.owner = c:CSH
				scope:target.owner = c:QNG
				scope:actor.unit_location = location:funing
			}
		}
		c:CSH = { trigger_event_silently = csh_1644_unification.6 }
	}
}
