on_dynasty_stability_monthly_check = {
    trigger = {
        has_variable = dynasty_stability
        NOT = { has_global_variable = temp_monthly_dynasty_stability_check}
    }
    effect = {
        set_global_variable = {
            name = temp_monthly_dynasty_stability_check
            days = 10
        }
        monthly_dynasty_stability_check_effect = yes
    }
}

on_chinese_warlord_monthly_check = {
    trigger = {
        NOT = { has_global_variable = temp_on_chinese_warlord_monthly_check}
    }
    effect = {
        every_country = {
            limit = {
                NOT = {
                    country_rank = country_rank:rank_empire
                }
                is_chinese_country = yes
                NOT = {
                    OR = {
                        has_reform = government_reform:the_chinese_emperor
                        has_reform = government_reform:the_chinese_warlords
                        has_reform = government_reform:fanwang_reform
                    }
                }
            }
            add_reform = government_reform:the_chinese_warlords
        }
        every_country = {
            limit = {
                country_rank = country_rank:rank_empire

                is_chinese_country = yes
                NOT = {
                    OR = {
                        has_reform = government_reform:the_chinese_emperor
                        has_reform = government_reform:the_chinese_warlords
                        has_reform = government_reform:fanwang_reform
                    }
                }
            }
            add_reform = government_reform:the_chinese_emperor
        }
        set_global_variable = {
            name = temp_on_chinese_warlord_monthly_check
            days = 10
        }
    }
}

on_late_ming_crisis_monthly = {
    trigger = {
        is_situation_active = situation:late_ming_crisis
        has_global_variable = ming_emperor_in_nanjing
        location:shangyuan = {
            OR = {
                owner != c:MNG
                controller != c:MNG
            }
        }
    }
    effect = {
        c:MNG = {
            trigger_event_non_silently = flavor_mng.2
        }
    }
}
