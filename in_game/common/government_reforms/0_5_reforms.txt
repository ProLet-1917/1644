the_chinese_emperor = {
    major = yes
    government = monarchy
    potential = {
        OR = {
            is_chinese_country = yes
            has_dlc = "0_5_dawn_of_the_new_era"
        }
        country_rank = country_rank:rank_empire
    }

    allow = {
        OR = {
            is_chinese_country = yes
            has_dlc = "0_5_dawn_of_the_new_era"
        }
        country_rank = country_rank:rank_empire
    }

    locked = {
        is_locked_mechanic = {
            mechanic = government_reform
            type = the_chinese_emperor
        }
    }

    on_activate = {
        remove_reform = government_reform:the_chinese_warlords
        if = {
            limit = {
                current_year >= 1645
                NOR = {
                    has_variable = claim_dynasty_name
                    this = c:CXI
                }
            }
            trigger_event_non_silently = flavor_all_chi.1
        }
        if = {
            limit = {
                NOT = {
                    is_member_of_international_organization = international_organization:chinese_dynasty
                }
            }
            create_international_organization = {
                type = international_organization_type:chinese_dynasty
                creator = prev
                add_country_to_international_organization = prev
                set_leader_country = prev
                add_policy_to_international_organization = policy:cd_fanzhen_shuwei
                add_policy_to_international_organization = policy:cd_establish_chaogong_system
            }
            every_subject_or_below = {
                international_organization:chinese_dynasty = {
                    add_country_to_international_organization = prev
                }
            }
        }
        else_if = {
            limit = {
                is_member_of_international_organization = international_organization:chinese_dynasty
                NOT = {
                    is_leader_of_international_organization = international_organization:chinese_dynasty
                }
            }
            international_organization:chinese_dynasty = {
                remove_country_from_international_organization = {
                    country = prev
                }
            }
            create_international_organization = {
                type = international_organization_type:chinese_dynasty
                creator = prev
                add_country_to_international_organization = prev
                set_leader_country = prev
                add_policy_to_international_organization = policy:cd_fanzhen_shuwei
                add_policy_to_international_organization = policy:cd_establish_chaogong_system
            }
            every_subject_or_below = {
                international_organization:chinese_dynasty = {
                    add_country_to_international_organization = prev
                }
            }
        }
        else_if = {
            is_leader_of_international_organization = international_organization:chinese_dynasty
        }
    }

    country_modifier = {
        allow_tributary_subject = yes
        monthly_prestige = 0.05
        government_reform_slots = 1
        global_max_bureaucracy_slots = 1
        cultures_capacity = 8
        dynasty_stability_monthly_income = 0.05
        allow_dynasty_stability = yes
    }

    years = 2
}

        }
    }

    allow = {
        OR = {

            has_reform = government_reform:the_chinese_emperor

        allow_dynasty_stability = yes
        enable_forced_dynamic_territorial_occupation = yes
    }

    years = 2
}
