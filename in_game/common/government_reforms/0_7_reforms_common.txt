# Anno 1644 - Age of Imperialism (age_7_imperialism)
# Generic government reforms available to (almost) every government type.
# Vanilla-style: one file per government type. This file holds the cross-government reforms.
#
# These reforms unlock as soon as the Age of Imperialism begins; they do not require
# a technology to be researched.

# 现代文官制度
modern_civil_service_reform = {
    age = age_7_imperialism

    potential = {
        OR = {
            always = yes
            has_dlc = "0_5_dawn_of_the_new_era"
        }
    }

    allow = {
        current_age_or_later = { age = age_7_imperialism }
    }

    country_modifier = {
        legislative_efficiency = 0.10
        global_max_bureaucracy_slots = 1
        country_cabinet_efficiency = 0.05
        monthly_towards_centralization = societal_value_monthly_move
        monthly_towards_innovative = societal_value_monthly_move
    }

    years = 2
}
