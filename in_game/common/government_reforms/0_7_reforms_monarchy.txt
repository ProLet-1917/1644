# Anno 1644 - Age of Imperialism (age_7_imperialism)
# Monarchy-only government reforms. Unlock automatically when the age begins.

# 君主立宪制
constitutional_monarchy_reform = {
    age = age_7_imperialism
    major = yes
    government = monarchy

    potential = {
        OR = {
            always = yes
            has_dlc = "0_5_dawn_of_the_new_era"
        }
        NOT = { is_revolutionary = yes }
    }

    allow = {
        current_age_or_later = { age = age_7_imperialism }
    }

    locked = {
        is_revolutionary = yes
    }

    on_activate = {
        set_parliament_type = parliament_type:imperial_parliament
    }

    country_modifier = {
        government_reform_slots = 1
        government_size = 1
        parliament_base_support = 0.15
        parliament_request_issue_support_needed = -0.10
        global_crown_estate_power = -0.10
        global_burghers_estate_power = 0.15
        global_nobles_estate_power = 0.05
        allow_imperial_parliament = yes
        monthly_towards_liberalism = societal_value_monthly_move
        monthly_towards_free_subjects = societal_value_monthly_move
    }

    years = 3
}
