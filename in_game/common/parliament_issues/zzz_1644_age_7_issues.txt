# Anno 1644 - Age of Imperialism
# Age 7 parliament issues. These are the law-making instrument of the new chambers:
# with a republican / imperial / presidential parliament the country may no longer simply
# pay stability for a policy change, so the issues below are how reform actually happens.
#
# Structure follows vanilla country parliament issues.

# 军队法案
army_bill = {
    estate = nobles_estate

    modifier_when_in_debate = {
        global_manpower_modifier = 0.05
    }

    allow = {
        OR = {
            always = yes
            has_dlc = "0_5_dawn_of_the_new_era"
        }
        current_age_or_later = { age = age_7_imperialism }
    }

    on_debate_passed = {
        add_country_modifier = {
            modifier = passed_army_bill
            years = 10
        }
        change_societal_value = { type = quality_vs_quantity value = societal_value_tiny_move_to_right }
    }

    on_debate_failed = {
        add_estate_satisfaction = {
            type = estate_type:nobles_estate
            value = estate_satisfaction_mild_penalty
        }
        parliament_debate_failed_effect = yes
    }

    chance = 6
}

# 殖民地自治法案
colonial_autonomy_bill = {
    estate = burghers_estate

    modifier_when_in_debate = {
        colonial_maintenance_efficiency = 0.10
    }

    allow = {
        OR = {
            always = yes
            has_dlc = "0_5_dawn_of_the_new_era"
        }
        current_age_or_later = { age = age_7_imperialism }
        OR = {
            is_colonial_overlord = yes
            any_subject = { is_subject_type = colonial_nation }
        }
    }

    on_debate_passed = {
        add_country_modifier = {
            modifier = passed_colonial_autonomy_bill
            years = 10
        }
        every_subject = {
            limit = { is_subject_type = colonial_nation }
            add_country_modifier = {
                modifier = granted_colonial_autonomy
                years = 10
            }
        }
    }

    on_debate_failed = {
        every_subject = {
            limit = { is_subject_type = colonial_nation }
            add_liberty_desire = 5
        }
        parliament_debate_failed_effect = yes
    }

    chance = 5
}
