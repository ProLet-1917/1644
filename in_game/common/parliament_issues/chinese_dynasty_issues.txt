# 王朝议会：确认继统、征藩镇兵、改土归流
cd_confirm_imperial_succession = {
    type = international_organization
    special_status = tianzi
    potential = {
        leader_country ?= {
            has_heir = yes
        }
    }
    allow = {
        NOT = { has_cooldown = cd_confirm_imperial_succession }
        leader_country ?= {
            has_heir = yes
        }
    }
    chance = 8
    selectable_for = {
        is_leader_of_international_organization = scope:recipient
    }
    on_debate_passed = {
        hidden_effect = {
            add_cooldown = {
                type = cd_confirm_imperial_succession
                years = 15
            }
        }
        leader_country ?= {
            change_dynasty_stability = { value = 50 }
            add_prestige = prestige_mild_bonus
            add_legitimacy = legitimacy_mild_bonus
        }
    }
    on_debate_failed = {
        custom_tooltip = {
            text = "lose_dynasty_stability_mild"
        }
        hidden_effect = {
            leader_country ?= {
                change_dynasty_stability = { value = -50 }
            }
        }
    }
    wants_this_parliament_issue_bias = {
        if = {
            limit = { scope:recipient = { international_organization_has_leader = yes } }
            add = {
                desc = "VOTE_BASE_VALUE"
                value = -200
            }
            add = {
                desc = "game_concept_opinion"
                value = {
                    value = "root.opinion(scope:recipient.leader_country)"
                    multiply = 0.5
                }
            }
        }
        else = {
            add = {
                desc = "VOTE_BASE_VALUE"
                value = 0
            }
        }
    }
}
