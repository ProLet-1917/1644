fanwang_shuwei = {
    get_payer_list = {
        leader_country ?= {
            add_to_local_variable_list = {
                name = payers
                target = this
            }
        }
    }
    get_payee_list = {
        every_international_organization_member = {
            limit = {
                NOT = { is_leader_of_international_organization = root }
                modifier:is_fanwang = yes
            }
            add_to_local_variable_list = {
                name = payees
                target = this
            }
        }
    }
    price = fanwang_shuwei_price
    uses_maintenance = yes
    min_slider_value = 0
    maintenance_modifier = {
        potential_trigger = { "payment_maintenance(scope:recipient|fanwang_shuwei)" < 0.5 }
        scale = {
            value = 1
            subtract = {
                value = "payment_maintenance(scope:recipient|fanwang_shuwei)"
                multiply = 2
            }
        }
        dynasty_stability_monthly_income = -0.025
    }
    maintenance_modifier = {
        potential_trigger = { "payment_maintenance(scope:recipient|fanwang_shuwei)" > 0.5 }
        scale = {
            value = "payment_maintenance(scope:recipient|fanwang_shuwei)"
            subtract = 0.5
            multiply = 2
        }
        dynasty_stability_monthly_income = 0.025
    }

    price_multiplier = {
        if = {
            #nothing happens during a power struggle
            limit = {
                international_organization_has_leader = no
            }
            value = 0
        }
        else = {
            every_international_organization_member = {
                limit = {
                    not = { is_leader_of_international_organization = root }
                }
                add = monthly_income_trade_and_tax
            }
            multiply = 0.01
        }
    }

    proportion_for_payer = {
        add = {
            desc = "game_concept_monthly_income_trade_and_tax"
            value = monthly_income_trade_and_tax
        }
    }

    proportion_for_payee = {
        add = {
            desc = "fanwang_shuwei"
            value = 1
        }
    }
}
