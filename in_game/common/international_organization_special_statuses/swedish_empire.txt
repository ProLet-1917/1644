dominion_of_empire = {
	priority = 5
	modifier = {
	}
	can_bestow_trigger = {
		is_member_of_international_organization = international_organization:swedish_empire
		NOT = { is_subject_of = international_organization:swedish_empire.leader_country }
		international_organization:swedish_empire = {
			NOT = {
				country_has_special_status = { type = special_status:kung_of_swedish country = root }
			}
			NOT = {
				country_has_special_status = { type = special_status:kingdom_of_finland country = root }
			}
			NOT = {
				country_has_special_status = { type = special_status:crownland_of_swedish country = root }
			}
			NOT = {
				country_has_special_status = { type = special_status:chartered_colony_of_swedish_empire country = root }
			}
		}
	}

	auto_bestowal_trigger = {
		always = yes
	}

	auto_rescind_trigger = {
		OR = {
			NOT = { is_member_of_international_organization = international_organization:swedish_empire }
			has_special_status_in_international_organization = { type = special_status:kung_of_swedish international_organization = international_organization:swedish_empire }
			has_special_status_in_international_organization = { type = special_status:crownland_of_swedish international_organization = international_organization:swedish_empire }
			has_special_status_in_international_organization = { type = special_status:chartered_colony_of_swedish_empire international_organization = international_organization:swedish_empire }
		}
	}

	on_bestowed_effect = {
	}

	on_rescinded_effect = {
	}
	#map_color =

	special_status_power = {
		value = 1
	}
}
