tributary = {
    priority = 1

    can_bestow_trigger = {
        is_member_of_international_organization = scope:recipient
        OR = {
            NOT = { is_subject_of = scope:recipient.leader_country }
            AND = {
                is_subject_of = scope:recipient.leader_country
                subject_type = subject_type:tributary
            }
        }
        scope:recipient = {
            NOT = {
                country_has_special_status = { type = special_status:tianzi country = root }
            }
            NOT = {
                country_has_special_status = { type = special_status:fanwang country = root }
            }
            NOT = {
                country_has_special_status = { type = special_status:fanzhen country = root }
            }
            NOT = {
                country_has_special_status = { type = special_status:tusi country = root }
            }
        }
    }

    auto_bestowal_trigger = {
        always = yes
    }

    auto_rescind_trigger = {
        OR = {
            NOT = { is_member_of_international_organization = scope:recipient }
            has_special_status_in_international_organization = { type = special_status:tianzi international_organization = scope:recipient }
            has_special_status_in_international_organization = { type = special_status:fanwang international_organization = scope:recipient }
            has_special_status_in_international_organization = { type = special_status:fanzhen international_organization = scope:recipient }
            has_special_status_in_international_organization = { type = special_status:tusi international_organization = scope:recipient }
        }
    }

    on_bestowed_effect = {
    }

    on_rescinded_effect = {

    }

    map_color = rgb { 130 157 108 }

    special_status_power = {
        value = 0
    }

    modifier = {
        prestige_decay = 0.01
        is_tributary = yes
    }
}
