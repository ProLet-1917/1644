# 宗法继承法（Patrilineal Primogeniture）
# 禁止女性继承，只允许男性继承人
# 必须从同宗族内选择继承者，优先选择直系血亲（亲生子女），如果没有则选择其他同宗族成员
# 基于 cognatic_primogeniture，但设置 allow_female = no 和 through_female = no

patrilineal_primogeniture = {
	include_ruler_siblings = yes
	all_in_dynasty = yes  # 搜索整个dynasty中的所有成员，不仅仅是家族树
	through_female = no
	allow_female = no
	allow_children = yes
	ignore_ruler = yes
	
	potential = {
		NOR = {
			succession_law = heir_selection:fratricide_succesion
			succession_law = heir_selection:unigeniture
		}
	}
	
	allowed_estates = { nobles_estate }
	
	heir_is_allowed = {
		# all_in_dynasty = yes 已经确保了只有同宗族成员会被考虑
		# 这里可以添加额外的限制（如果需要）
	}
	
	traverse_family_tree = yes
	
	calc = {
		# 使用calc而不是sibling_score，因为all_in_dynasty可能需要calc
		if = {
			limit = {
				exists = root
			}
			# 直系血亲（亲生儿子）最高分，优先选择
			if = {
				limit = {
					# 必须是统治者的亲生儿子（父系血统，男性）
					exists = father
					any_parent = {
						this = scope:target.ruler
					}
					is_female = no
				}
				add = {
					desc = "HEIR_SELECTION_DIRECT_DESCENDANT"
					value = -1000  # 大幅减分，确保亲儿子绝对优先（分数越低越优先）
				}
			}
			
			# 兄弟姐妹（第二优先级）
			if = {
				limit = {
					# 不是直系血亲（避免重复加分）
					NOT = {
						exists = father
						any_parent = {
							this = scope:target.ruler
						}
						is_female = no
					}
					# 有共同的父亲（同父兄弟，父系优先）
					exists = father
					scope:target.ruler.exists = father
					father = scope:target.ruler.father
					is_female = no
				}
				add = {
					desc = "HEIR_SELECTION_SIBLING"
					value = -500  # 兄弟姐妹减分，但远高于亲儿子（分数越低越优先）
				}
			}
			
			# 叔伯/侄子（第三优先级）
			if = {
				limit = {
					# 不是直系血亲或兄弟姐妹（避免重复加分）
					NOT = {
						OR = {
							AND = {
								exists = father
								any_parent = {
									this = scope:target.ruler
								}
								is_female = no
							}
							AND = {
								exists = father
								scope:target.ruler.exists = father
								father = scope:target.ruler.father
								is_female = no
							}
						}
					}
					# 是统治者的侄子（兄弟的儿子，父系）
					OR = {
						AND = {
							exists = father
							father.exists = father
							father.father = scope:target.ruler.father
							is_female = no
						}
						# 是统治者的叔伯（父亲的兄弟，父系）
						AND = {
							scope:target.ruler.exists = father
							exists = father
							father.exists = father
							father.father = scope:target.ruler.father.father
							NOT = {
								this = scope:target.ruler.father
							}
							is_female = no
						}
					}
				}
				add = {
					desc = "HEIR_SELECTION_UNCLE_OR_NEPHEW"
					value = -200  # 叔伯/侄子减分，但远高于兄弟姐妹（分数越低越优先）
				}
			}
		
			# 其他同宗族成员（表兄弟等远亲，第四优先级）
			if = {
				limit = {
					# 不是直系血亲、兄弟姐妹或叔伯/侄子（避免重复加分）
					NOT = {
						OR = {
							AND = {
								exists = father
								any_parent = {
									this = scope:target.ruler
								}
								is_female = no
							}
							AND = {
								exists = father
								scope:target.ruler.exists = father
								father = scope:target.ruler.father
								is_female = no
							}
							AND = {
								exists = father
								father.exists = father
								father.father = scope:target.ruler.father
								is_female = no
							}
							AND = {
								scope:target.ruler.exists = father
								exists = father
								father.exists = father
								father.father = scope:target.ruler.father.father
								NOT = {
									this = scope:target.ruler.father
								}
								is_female = no
							}
						}
					}
					# 必须是男性
					is_female = no
				}
				add = {
					desc = "HEIR_SELECTION_SAME_DYNASTY"
					value = -50  # 其他同宗族成员基础分，远高于近亲（分数越低越优先）
				}
			}
			
			# 年长继承人减分（使用root.character_age，参考原版实现）
			# 注意：年龄减分应该很小，不能影响亲疏关系的优先级
			# 年龄越大分数越低（越优先）
			add = {
				desc = "HEIR_SELECTION_OLDER_HEIR_DESC"
				value = -root.character_age
			}
			
			# 不适合统治的继承人加分（分数越高越不优先）
			if = { 
				limit = { modifier:blocked_from_being_ruler = yes }
				add = {
					desc = "HEIR_SELECTION_IS_UNSUITED_TO_RULE_DESC"
					value = 1000  # 大幅加分，确保不适合统治的人排在后面
				}
			}
		}
		else = {
			# 描述（当不存在root时）
			add = {
				desc = "HEIR_SELECTION_OLDER_HEIR_DESC"
				value = 1
			}
		}
	}
}
