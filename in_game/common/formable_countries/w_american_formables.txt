# Anno 1644 - Formable nations of the New World
#
# These allow the colonial nations created by the 1644 setup (NEN, NFR, NSP, VOP, VOB)
# and any other colony or former colony in the same region to be founded as a nation
# in their own right. Flags are defined in main_menu/common/flag_definitions/w_american_flags.txt.

# 巴西
VOB_f = { # Brazil
    level = 3
    required_locations_fraction = 0.5
    rule = historical

    capital_required = yes

    potential = {
        capital ?= { region = region:brazil_region }
    }

    allow = {
        is_subject = no
    }

    name = VOB
    flag = VOB
    adjective = VOB_ADJ
    tag = VOB
    color = map_VOB

    regions = {
        brazil_region
    }

    form_effect = {
        if = {
            limit = { country_rank_level < 3 }
            set_country_rank_effect = { rank = country_rank:rank_kingdom }
        }
        add_country_modifier = {
            modifier = new_world_nation_founded
            years = 25
        }
    }
}
