# Anno 1644 - Formable nations created by colonial independence
#
# Companion to w_american_formables.txt. Every nation that historically won its
# independence from a colonial power has a formable here, so that a released or
# victorious colony can constitute itself as a sovereign state.
# Flags are defined in main_menu/common/flag_definitions/w_colonial_flags.txt.

#############################################
# Spanish America
#############################################

GRN_f = { # New Granada
    level = 3
    required_locations_fraction = 0.3
    rule = historical
    capital_required = yes

    potential = { capital ?= { region = region:colombia_region } }
    allow = { is_subject = no }

    name = GRN
    flag = GRN
    adjective = GRN_ADJ
    tag = GRN
    color = map_GRN

    regions = { colombia_region }

    form_effect = {
        if = { limit = { country_rank_level < 3 } set_country_rank_effect = { rank = country_rank:rank_kingdom } }
        add_country_modifier = { modifier = new_world_nation_founded years = 25 }
    }
}
