on_plundering_value = {
    value = 1
    multiply = {
        value = define:NAnno1644|POP_GOT_PLUNDER_BASIC
        #value = scope:new_controller.modifier:population_got_plunder_basic # 1.6% * 1000 = 16 People
    }
    if = {
        limit = {
            scope:character.army_size <= define:NAnno1644|POP_GOT_PLUNDER_REQUIRE
        }
        divide = {
            value = scope:character.army_size
            divide = 20
        }
    }
    if = {
        limit = {
            OR = {
                this.culture = scope:new_controller.culture
                this.religion = scope:new_controller.religion
            }
        }
        multiply = 0.5
    }
    if = {
        limit = {
            AND = {
                this.culture = scope:new_controller.culture
                this.religion = scope:new_controller.religion
            }
        }
        multiply = 0.5 #0.5*0.5 = 0.25
    }
    multiply = -1
}
