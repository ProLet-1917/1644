chi_reintegrate_area = {
    ability = mil
    allow_multiple = yes

    potential = {
        capital ?= {
            is_in_scripted_geography = scripted_geography:china_geography
        }
    }

    allow = {
        custom_tooltip = {
            text = location_located_in_china_region
            any_province = {
                any_location_in_province = {
                    is_in_scripted_geography = scripted_geography:china_geography
                    integration_level = conquered
                }
            }
        }
    }

    select_trigger = {
        looking_for_a = area
        source = actor
        target_flag = target
        name = "integrate_area_select_area"
        none_available_msg_key = "integrate_area_no_areas"
        column = {
            data = name
        }
        column = {
            data = integration
        }
        visible = {
            custom_tooltip = {
                text = location_located_in_china_region
                any_location_in_area = {
                    is_in_scripted_geography = scripted_geography:china_geography
                    owner ?= scope:actor
                    integration_level = conquered
                }
                NOT = { cabinet_already_performing_same_task_on_target = { cabinet_action = integrate_area interaction_target = target } }
            }
        }
        map_color = {
            lerp = {
                min_color = define:NMapColors|MAP_COLOR_LOW
                max_color = define:NMapColors|MAP_COLOR_HIGH
                factor = integration_progress
            }
        }
    }

    is_finished = {
        scope:target = {
            not = {
                any_location_in_area = {
                    owner ?= scope:actor
                    OR = {
                        integration_level = conquered
                        integration_level = colonized
                    }
                    count > 0
                }
            }
        }
    }

    province_modifier = {
        province_integration_speed = 2.5
    }

    progress = {
        scope:target = {
            value = "area_average_integration(scope:actor)"
        }
    }

    map_marker = {
        show_for_owner = yes
    }


}
