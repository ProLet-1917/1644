namespace = the_steppe_random_event

the_steppe_random_event.1 = {
    type = country_event

    title = the_steppe_random_event.1.title
    desc = the_steppe_random_event.1.desc

    trigger = {
        has_variable = current_herd
        government_type = government_type:steppe_horde
    }

    option = {
        name = the_steppe_random_event.1.a
        custom_tooltip = {
            text = the_steppe_random_event.1.tt
            change_variable = {
                name = current_herd
                add = 50
            }
        }
    }
}
