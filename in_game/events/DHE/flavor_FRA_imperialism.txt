# 普法战争
flavor_fra_imperialism.4 = {
    type = country_event
    fire_only_once = yes
    title = flavor_fra_imperialism.4.title
    desc = flavor_fra_imperialism.4.desc
    outcome = negative

    historical_info = flavor_fra_imperialism.4.historical_info

    dynamic_historical_event = {
        tag = FRA
        from = 1866.1.1
        to = 1885.1.1
        monthly_chance = 3
    }

    trigger = {
        OR = {
            always = yes
            has_dlc = "0_5_dawn_of_the_new_era"
        }
        OR = {
            country_exists = c:PRU
            country_exists = c:GER
        }
    }

    illustration_tags = {
        10 = exterior
        10 = angry
    }

    immediate = {
        event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:nobles_estate }
        if = {
            limit = { country_exists = c:GER }
            c:GER = { save_scope_as = german_rival }
        }
        else_if = {
            limit = { country_exists = c:PRU }
            c:PRU = { save_scope_as = german_rival }
        }
    }

    option = {
        # 迎战
        name = flavor_fra_imperialism.4.a
        historical_option = yes
        ai_will_select = { base = 60 }
        add_stability = stability_severe_penalty
        add_prestige = prestige_severe_penalty
        if = {
            limit = { exists = scope:german_rival }
            add_casus_belli = {
                target = scope:german_rival
                type = casus_belli:cb_conquer_province
            }
        }
        add_country_modifier = { modifier = revanchism_against_germany years = -1 }
    }

    option = {
        # 避战
        name = flavor_fra_imperialism.4.b
        ai_will_select = { base = 40 }
        add_stability = stability_mild_penalty
        add_prestige = prestige_mild_penalty
    }
}

            culture = culture:french
            religion = religion:catholic

        name = flavor_fra_imperialism.10.a

            move_country = root

            add_cultural_tradition = cultural_tradition_mild_bonus

            kill_character_silently = scope:impressionist
