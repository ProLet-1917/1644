# 克里米亚战争
flavor_rus_imperialism.1 = {
    type = country_event
    fire_only_once = yes

    desc = flavor_rus_imperialism.1.desc
    outcome = negative

    historical_info = flavor_rus_imperialism.1.historical_info

    dynamic_historical_event = {
        tag = RUS
        tag = MOS
        from = 1850.1.1
        to = 1870.1.1
        monthly_chance = 4
    }

    trigger = {
        OR = {
            always = yes
            has_dlc = "0_5_dawn_of_the_new_era"
        }
        has_presence_in = region:ruthenia_region
    }

    illustration_tags = {
        10 = exterior
        10 = angry
    }

    immediate = {
        event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:nobles_estate }
    }

    option = {
        name = flavor_rus_imperialism.1.a

        add_country_modifier = { modifier = crimean_war_humiliation years = 20 }
        add_stability = stability_severe_penalty
        add_prestige = prestige_severe_penalty
        change_societal_value = { type = traditionalist_vs_innovative value = societal_value_tiny_move_to_right }

        name = flavor_rus_imperialism.1.b
        add_stability = stability_mild_penalty
        add_prestige = prestige_mild_penalty
        change_societal_value = { type = outward_vs_inward value = societal_value_tiny_move_to_left }
    }

# 俄罗斯文学
flavor_rus_imperialism.10 = {
    hide_portraits = yes
    type = country_event
    fire_only_once = yes
    title = flavor_rus_imperialism.10.title
    desc = flavor_rus_imperialism.10.desc
    outcome = neutral

    historical_info = flavor_rus_imperialism.10.historical_info

    dynamic_historical_event = {
        tag = RUS
        tag = MOS
        from = 1850.1.1
        to = 1900.1.1
        monthly_chance = 2
    }

    trigger = {
        OR = {
            always = yes
            has_dlc = "0_5_dawn_of_the_new_era"
        }
        OR = {
            has_advance = modern_education_advance
            current_age_or_later = { age = age_7_imperialism }
        }
    }

    illustration_tags = {
        10 = happy
        10 = interior
    }

    immediate = {
        event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:nobles_estate }
        create_character = {
            adm = { 30 90 }
            dip = { 30 90 }
            mil = { 10 50 }
            artist = artist_type:writer
            artist_skill = { 0.5 1.0 }
            birth_date = 1828.9.9
            female = no
            culture = root.culture
            religion = root.religion
            estate = estate_type:nobles_estate
            create_in_limbo = yes
            save_scope_as = russian_writer
        }
    }

    option = {
        name = flavor_rus_imperialism.10.a
        historical_option = yes
        scope:russian_writer = {
            move_country = root
            add_character_modifier = {
                modifier = russian_man_of_letters
                years = -1
            }
        }
        culture ?= {
            add_cultural_influence = cultural_influence_mild_bonus
            add_cultural_tradition = cultural_tradition_mild_bonus
        }
        add_research_progress = research_progress_mild_bonus
    }

    option = {
        name = flavor_rus_imperialism.10.b
        add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_mild_bonus }
        change_societal_value = { type = traditionalist_vs_innovative value = societal_value_tiny_move_to_left }
    }
}
