# 西班牙艺术
flavor_spa_imperialism.10 = {
    hide_portraits = yes
    type = country_event
    fire_only_once = yes
    title = flavor_spa_imperialism.10.title
    desc = flavor_spa_imperialism.10.desc
    outcome = neutral

    historical_info = flavor_spa_imperialism.10.historical_info

    dynamic_historical_event = {
        tag = SPA
        tag = CAS
        from = 1870.1.1
        to = 1910.1.1
        monthly_chance = 2
    }

    trigger = {
        OR = {
            always = yes
            has_dlc = "0_5_dawn_of_the_new_era"
        }
        current_age_or_later = { age = age_6_revolutions }
    }

    illustration_tags = {
        10 = happy
        10 = interior
    }

    immediate = {
        event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:burghers_estate }
        create_character = {
            adm = { 30 90 }
            dip = { 30 90 }
            mil = { 10 50 }
            artist = artist_type:architect
            artist_skill = { 0.5 1.0 }
            birth_date = 1852.6.25
            female = no
            culture = root.culture
            religion = root.religion
            estate = estate_type:burghers_estate
            create_in_limbo = yes
            save_scope_as = catalan_architect
        }
    }

    option = {
        name = flavor_spa_imperialism.10.a
        historical_option = yes
        scope:catalan_architect = {
            move_country = root
            add_character_modifier = {
                modifier = catalan_modernisme_architect
                years = -1
            }
        }
        culture ?= {
            add_cultural_influence = cultural_influence_mild_bonus
            add_cultural_tradition = cultural_tradition_mild_bonus
        }
        add_prestige = prestige_mild_bonus
    }

    option = {
        name = flavor_spa_imperialism.10.b
        hidden_effect = {
            kill_character_silently = scope:catalan_architect
        }
        add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_mild_bonus }
    }
}
