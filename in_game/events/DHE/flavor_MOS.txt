# The Arkhangelsk Decree
flavor_mos.59 = {
	type = country_event
	fire_only_once = yes
	title = flavor_mos.59.title
	desc = flavor_mos.59.desc
	outcome = neutral

	dynamic_historical_event = {
		tag = RUS
		from = 1644.1.1
		to = 1700.1.1
		monthly_chance = 5
	}

	trigger = {

		location:arkhangelsk = {
			owner = root
			controller = root
			NOT = {
				has_building = building_type:port_authority
			}
		}
		area:arkhangelsk_area = {
			any_location_in_area = {
				count < 3
				OR = {
					owner = root
					owner ?= { top_overlord = root }
				}
				is_coastal = yes
			}
		}

	}

	immediate = {

		location:arkhangelsk = {
			save_scope_as = target_location
		}

		root = { save_scope_as = target_root_country }

	}

	option = {
		name = flavor_mos.59.a
		historical_option = yes

		change_gold_effect = { scale = -3 }

		location:arkhangelsk = {
			if = {
				limit = {
					NOR = {
						location_rank = location_rank:town
						location_rank = location_rank:city
					}
				}
				change_location_rank_effect = { location_rank = location_rank:town }
			}
			construct_building = {
				building_type = building_type:port_authority
			}
		}

		add_migration = {
			owner = root
			from = capital.province_definition
			to = scope:target_location.province_definition
			type = pop_type:peasants
			amount = 0.05
			months = 12
		}

		add_country_modifier = {
			modifier = bustling_trade_hub
			years = 10
			mode = add_and_extend
		}

		add_estate_satisfaction = { type = estate_type:burghers_estate
			value = estate_satisfaction_severe_bonus
		}

		custom_tooltip = flavor_mos.59.tt1

		ai_chance = {
			factor = 1
		}
	}

	option = {
		name = flavor_mos.59.b

		nudge_towards_isolationism = yes

		add_prestige = prestige_mild_bonus

		ai_chance = {
			factor = 0.5
		}
	}

	historical_info = flavor_mos.59.historical_info

}
