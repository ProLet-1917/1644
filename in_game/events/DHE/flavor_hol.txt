#Synod of Dort
flavor_hol.2 = {
	type = country_event
	title = flavor_hol.2.title
	desc = flavor_hol.2.desc
	outcome = neutral
	historical_info = flavor_hol.2.historical_info
	fire_only_once = yes
	dynamic_historical_event = {
		tag = NED
		from = 1644.1.1
		to = 1684.1.1
		monthly_chance = 2
	}
	trigger = {
		owns = location:dordrecht
		religion:calvinist = {
			is_religion_enabled = yes
			has_unlocked_religious_aspect_trigger = { type = gomarism }
			has_unlocked_religious_aspect_trigger = { type = arminianism }
		}
		OR = {
			has_religious_aspect = religious_aspect:gomarism
			has_religious_aspect = religious_aspect:arminianism
		}
	}
	illustration_tags = {
		10 = angry
		10 = interior
	}
	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:clergy_estate background = estate_type:clergy_estate }
		location:dordrecht = { save_scope_as = target_location }
		character:jacobus_arminius = { save_scope_as = loc_jacobus_arminius }
		character:franciscus_gomarus = { save_scope_as = loc_franciscus_gomarus }
	}
	option = {
		name = flavor_hol.2.a
		historical_option = yes
		if = {
			limit = { has_religious_aspect = religious_aspect:arminianism }
			remove_religious_aspect = religious_aspect:arminianism
		}
		custom_tooltip = {
			text = flavor_hol.2.a.tt
			set_variable = { name = blocked_religious_aspect_arminianism value = yes }
			set_variable = { name = unlocked_instituionalized_gomarism value = yes }
			if = {
				limit = { has_religious_aspect = religious_aspect:gomarism }
				remove_religious_aspect = religious_aspect:gomarism
			}
			add_religious_aspect = religious_aspect:instituionalized_gomarism
		}
	}
	option = {
		name = flavor_hol.2.b
		if = {
			limit = { has_religious_aspect = religious_aspect:gomarism }
			remove_religious_aspect = religious_aspect:gomarism
		}
		custom_tooltip = {
			text = flavor_hol.2.b.tt
			set_variable = { name = blocked_religious_aspect_gomarism value = yes }
			set_variable = { name = unlocked_instituionalized_arminianism value = yes }
			if = {
				limit = { has_religious_aspect = religious_aspect:arminianism }
				remove_religious_aspect = religious_aspect:arminianism
			}
			add_religious_aspect = religious_aspect:instituionalized_arminianism
		}
	}
	option = {
		name = flavor_hol.2.c
		add_stability = stability_extreme_bonus
	}
}

	}
	option = {
		name = flavor_hol.18.a

			trigger_event_silently = { id = flavor_hol.19 }
