# Anno 1644 - Sweden in the Age of Imperialism (1837-1918)

namespace = flavor_swe_imperialism

# 失去芬兰后的重整
flavor_swe_imperialism.1 = {
    type = country_event
    fire_only_once = yes
    title = flavor_swe_imperialism.1.title
    desc = flavor_swe_imperialism.1.desc
    outcome = neutral

    historical_info = flavor_swe_imperialism.1.historical_info

    dynamic_historical_event = {
        tag = SWE
        from = 1815.1.1
        to = 1870.1.1
        monthly_chance = 4
    }

    trigger = {
        OR = {
            always = yes
            has_dlc = "0_5_dawn_of_the_new_era"
        }
        current_age_or_later = { age = age_6_revolutions }
    }

    illustration_tags = {
        10 = interior
        10 = regular
    }

    immediate = {
        event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:burghers_estate }
    }

    option = {
        name = flavor_swe_imperialism.1.a
        historical_option = yes
        add_country_modifier = { modifier = swedish_era_of_liberty years = 25 }
        add_stability = stability_mild_bonus
        change_societal_value = { type = absolutism_vs_liberalism value = societal_value_tiny_move_to_right }
        add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_mild_bonus }
    }

    option = {
        name = flavor_swe_imperialism.1.b
        add_country_modifier = { modifier = swedish_scandinavian_union years = 25 }
        add_prestige = prestige_mild_bonus
        change_societal_value = { type = outward_vs_inward value = societal_value_tiny_move_to_left }
    }
}

# 北欧艺术与科学
flavor_swe_imperialism.10 = {
    hide_portraits = yes
    type = country_event
    fire_only_once = yes
    title = flavor_swe_imperialism.10.title
    desc = flavor_swe_imperialism.10.desc
    outcome = neutral

    historical_info = flavor_swe_imperialism.10.historical_info

    dynamic_historical_event = {
        tag = SWE
        from = 1870.1.1
        to = 1915.1.1
        monthly_chance = 2
    }

    trigger = {
        OR = {
            always = yes
            has_dlc = "0_5_dawn_of_the_new_era"
        }
        current_age_or_later = { age = age_6_revolutions }
    }

    illustration_tags = {
        10 = happy
        10 = interior
    }

    immediate = {
        event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:clergy_estate }
        create_character = {
            adm = { 30 90 }
            dip = { 30 90 }
            mil = { 10 50 }
            artist = artist_type:writer
            artist_skill = { 0.5 1.0 }
            birth_date = 1849.3.20
            female = no
            culture = root.culture
            religion = root.religion
            estate = estate_type:burghers_estate
            create_in_limbo = yes
            save_scope_as = swedish_writer
        }
    }

    option = {
        name = flavor_swe_imperialism.10.a
        historical_option = yes
        scope:swedish_writer = {
            move_country = root
            add_character_modifier = {
                modifier = swedish_man_of_letters
                years = -1
            }
        }
        culture ?= {
            add_cultural_influence = cultural_influence_mild_bonus
            add_cultural_tradition = cultural_tradition_mild_bonus
        }
        add_research_progress = research_progress_mild_bonus
    }

    option = {
        name = flavor_swe_imperialism.10.b
        add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_mild_bonus }
        change_societal_value = { type = traditionalist_vs_innovative value = societal_value_tiny_move_to_left }
    }
}
