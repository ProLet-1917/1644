# 德国音乐与哲学
flavor_ger_imperialism.10 = {
    hide_portraits = yes
    type = country_event
    fire_only_once = yes
    title = flavor_ger_imperialism.10.title
    desc = flavor_ger_imperialism.10.desc
    outcome = neutral

    historical_info = flavor_ger_imperialism.10.historical_info

    dynamic_historical_event = {
        tag = GER
        from = 1840.1.1
        to = 1910.1.1
        monthly_chance = 2
    }

    trigger = {
        tag = GER
    }

    illustration_tags = {
        10 = happy
        10 = interior
    }

    immediate = {
        event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:clergy_estate }
        create_character = {
            adm = { 30 90 }
            dip = { 30 90 }
            mil = { 10 50 }
            artist = artist_type:composer
            artist_skill = { 0.6 1.0 }
            birth_date = 1813.5.22
            female = no
            culture = root.culture
            religion = root.religion
            estate = estate_type:burghers_estate
            create_in_limbo = yes
            save_scope_as = german_composer
        }
    }

    option = {
        name = flavor_ger_imperialism.10.a
        historical_option = yes
        scope:german_composer = {
            move_country = root
            add_character_modifier = {
                modifier = german_master_composer
                years = -1
            }
        }
        culture ?= {
            add_cultural_influence = cultural_influence_mild_bonus
            add_cultural_tradition = cultural_tradition_mild_bonus
        }
        add_prestige = prestige_mild_bonus
    }

    option = {
        name = flavor_ger_imperialism.10.b
        hidden_effect = {
            kill_character_silently = scope:german_composer
        }
        add_stability = stability_mild_bonus
    }
}
