# Anno 1644 - England in the Age of Imperialism (1837-1918)
# Flavor / Dynamic Historical Events for the United Kingdom
# Covers the era, plus the people and artworks of the age.

namespace = flavor_eng_imperialism

# The Chartist Movement
flavor_eng_imperialism.1 = {
	hide_portraits = yes
	type = country_event
	fire_only_once = yes

	title = flavor_eng_imperialism.1.title
	desc = flavor_eng_imperialism.1.desc
	outcome = neutral

	dynamic_historical_event = {
		tag = ENG
		tag = GBR
		from = 1839.1.1
		to = 1858.1.1
		monthly_chance = 2
	}

	trigger = {
		government_type = government_type:monarchy
	}

	illustration_tags = {
		10 = interior
		10 = angry
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:nobles_estate }
	}

	option = {
		name = flavor_eng_imperialism.1.a
		historical_option = yes
		change_societal_value = { type = aristocracy_vs_plutocracy value = societal_value_move_to_right }
		change_societal_value = { type = serfdom_vs_free_subjects value = societal_value_move_to_right }
		add_estate_satisfaction = { type = estate_type:peasants_estate value = estate_satisfaction_mild_bonus }
		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_mild_bonus }
	}

	option = {
		name = flavor_eng_imperialism.1.b
		add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_mild_bonus }
		change_societal_value = { type = serfdom_vs_free_subjects value = societal_value_move_to_left }
		add_stability = stability_mild_bonus
	}

	historical_info = flavor_eng_imperialism.1.historical_info
}

# The Great Famine in Ireland
flavor_eng_imperialism.2 = {
	hide_portraits = yes
	type = country_event
	fire_only_once = yes

	title = flavor_eng_imperialism.2.title
	desc = flavor_eng_imperialism.2.desc
	outcome = negative

	dynamic_historical_event = {
		tag = ENG
		tag = GBR
		from = 1845.1.1
		to = 1856.1.1
		monthly_chance = 3
	}

	trigger = {
		owns = location:dublin
	}

	illustration_tags = {
		10 = exterior
		10 = angry
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:peasants_estate background = estate_type:clergy_estate }
		every_owned_location = {
			limit = {
				region = region:ireland_region
				raw_material = goods:potato
			}
			add_location_modifier = {
				modifier = irish_potato_blight
				years = 10
				mode = add_and_extend
			}
		}
	}

	option = {
		name = flavor_eng_imperialism.2.a
		add_gold = -250
		add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_mild_bonus }
		add_estate_satisfaction = { type = estate_type:peasants_estate value = estate_satisfaction_mild_bonus }
	}

	option = {
		name = flavor_eng_imperialism.2.b
		historical_option = yes
		change_societal_value = { type = mercantilism_vs_free_trade value = societal_value_move_to_right }
		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_mild_bonus }
		add_stability = stability_mild_penalty
	}

	historical_info = flavor_eng_imperialism.2.historical_info
}

# The Pre-Raphaelite Brotherhood
flavor_eng_imperialism.5 = {
	hide_portraits = yes
	type = country_event
	fire_only_once = yes

	title = flavor_eng_imperialism.5.title
	desc = flavor_eng_imperialism.5.desc
	outcome = neutral

	dynamic_historical_event = {
		tag = ENG
		tag = GBR
		from = 1848.1.1
		to = 1855.1.1
		monthly_chance = 2
	}

	trigger = {
	    OR = {
	        always = yes
	        has_dlc = "0_5_dawn_of_the_new_era"
	    }
	    current_age_or_later = { age = age_6_revolutions }
	    stability < 40
	}

	illustration_tags = {
		10 = interior
		10 = happy
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:clergy_estate }
		create_character = {
			first_name = name_gabriel
			last_name = rossetti
			estate = estate_type:burghers_estate
			adm = 65
			dip = 82
			mil = 35
			birth_date = 1829.7.12
			birth_location = location:london
			culture = culture:english
			religion = religion:anglican
			artist = painter
			artist_skill = 0.75
			save_scope_as = target_character
		}
		scope:target_character = {
			add_character_modifier = {
				modifier = pre_raphaelite_master
				years = -1
				mode = add_and_extend
			}
		}
		capital = {
			create_art = {
				type = work_of_art_type:painting
				artist = scope:target_character
				quality = 8
			}
		}
	}

	option = {
		name = flavor_eng_imperialism.5.a

		add_prestige = prestige_mild_bonus

		name = flavor_eng_imperialism.5.b

	historical_info = flavor_eng_imperialism.5.historical_info
}
