namespace = flavor_plc_imperialism

# 十一月起义与民族之春
flavor_plc_imperialism.1 = {
    type = country_event
    fire_only_once = yes
    title = flavor_plc_imperialism.1.title
    desc = flavor_plc_imperialism.1.desc
    outcome = negative

    historical_info = flavor_plc_imperialism.1.historical_info

    dynamic_historical_event = {
        tag = PLC
        tag = POL
        tag = LIT
        from = 1830.1.1
        to = 1870.1.1
        monthly_chance = 4
    }

    trigger = {
        OR = {
            always = yes
            has_dlc = "0_5_dawn_of_the_new_era"
        }
        current_age_or_later = { age = age_6_revolutions }
    }

    illustration_tags = {
        10 = exterior
        10 = angry
    }

    immediate = {
        event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:peasants_estate }
    }

    option = {
        name = flavor_plc_imperialism.1.a
        historical_option = yes
        add_country_modifier = { modifier = polish_national_rising years = 20 }
        add_stability = stability_severe_penalty
        change_societal_value = { type = serfdom_vs_free_subjects value = societal_value_tiny_move_to_right }
        add_estate_satisfaction = { type = estate_type:peasants_estate value = estate_satisfaction_mild_bonus }
    }

    option = {
        name = flavor_plc_imperialism.1.b
        add_stability = stability_mild_bonus
        add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_mild_bonus }
        add_estate_satisfaction = { type = estate_type:peasants_estate value = estate_satisfaction_mild_penalty }
    }
}

# 波兰文化与音乐
flavor_plc_imperialism.10 = {
    hide_portraits = yes
    type = country_event
    fire_only_once = yes
    title = flavor_plc_imperialism.10.title
    desc = flavor_plc_imperialism.10.desc
    outcome = neutral

    historical_info = flavor_plc_imperialism.10.historical_info

    dynamic_historical_event = {
        tag = PLC
        tag = POL
        tag = LIT
        from = 1830.1.1
        to = 1900.1.1
        monthly_chance = 2
    }

    trigger = {
        OR = {
            always = yes
            has_dlc = "0_5_dawn_of_the_new_era"
        }
        current_age_or_later = { age = age_6_revolutions }
    }

    illustration_tags = {
        10 = happy
        10 = interior
    }

    immediate = {
        event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:nobles_estate }
        create_character = {
            adm = { 20 80 }
            dip = { 20 80 }
            mil = { 10 50 }
            artist = artist_type:composer
            artist_skill = { 0.6 1.0 }
            birth_date = 1810.3.1
            female = no
            culture = root.culture
            religion = root.religion
            estate = estate_type:burghers_estate
            create_in_limbo = yes
            save_scope_as = polish_composer
        }
    }

    option = {
        name = flavor_plc_imperialism.10.a
        historical_option = yes
        scope:polish_composer = {
            move_country = root
            add_character_modifier = {
                modifier = polish_master_composer
                years = -1
            }
        }
        culture ?= {
            add_cultural_influence = cultural_influence_mild_bonus
            add_cultural_tradition = cultural_tradition_mild_bonus
        }
        add_prestige = prestige_mild_bonus
    }

    option = {
        name = flavor_plc_imperialism.10.b
        hidden_effect = {
            kill_character_silently = scope:polish_composer
        }
        add_stability = stability_mild_bonus
    }
}
