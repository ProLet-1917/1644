namespace = flavor_csh
flavor_csh.1 = {
	#大同反正
	type = country_event
	title = flavor_csh.1.title
	desc = flavor_csh.1.desc
	historical_info = flavor_csh.1.historical_info
	trigger = {
		c:CJX ?= {
			is_subject_of = c:CSH
		}
		c:CSH ?= {
			has_variable = chi_domination
			var:chi_domination < 70			#军威
			NOT = {
				owns = location:funing
			}			#失去山海关
		}
		#
		c:WSG ?= {
			NOT = {
				is_subject_of = c:CSH				#吴三桂不是大顺附庸
			}
		}
		OR = {
			c:QNG.var:chi_domination > c:CSH.var:chi_domination			#清国军威大于大顺军威
			c:MNG.var:chi_domination > c:CSH.var:chi_domination			#明国军威大于大顺军威
		}
	}
	immediate = {
		c:CJX = {
			set_variable = cjx_rebellion_first
		}
	}

	fire_only_once = yes

	dynamic_historical_event = {
		tag = CSH
		from = 1644.6.1		# The game will check if this event can be fired after this date
		to = 1700.1.1		# The game will stop checking after this date.
		monthly_chance = 100		# Monthly chance that this event will fire (10 = 10% chance)
	}
	option = {
		#降清
		name = flavor_csh.1.a
		trigger = {
		    c:QNG.var:chi_domination > c:CSH.var:chi_domination
		    c:QNG.var:chi_domination > c:MNG.var:chi_domination
		}
		cancel_subject = c:CJX
		c:CJX = {
			make_subject_of = {
				target = c:QNG
				type = subject_type:cn_march_governor
			}
		}
	}
	option = {#归明
	    name = flavor_csh.1.b
	    trigger = {
	        c:MNG.var:chi_domination > c:CSH.var:chi_domination
	        c:MNG.var:chi_domination > c:QNG.var:chi_domination
	    }
	    c:CSH = {
	        cancel_subject = c:CJX
	    }
	    c:CJX = {
	        make_subject_of = {
	            target = c:MNG
	            type = subject_type:cn_march_governor
	        }
	    }
	}
}
