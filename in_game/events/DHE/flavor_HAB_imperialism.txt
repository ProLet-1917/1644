namespace = flavor_hab_imperialism

# 1848年革命
flavor_hab_imperialism.1 = {
    type = country_event
    fire_only_once = yes
    title = flavor_hab_imperialism.1.title
    desc = flavor_hab_imperialism.1.desc
    outcome = negative

    historical_info = flavor_hab_imperialism.1.historical_info

    dynamic_historical_event = {
        tag = HAB
        tag = AUS
        from = 1848.3.1
        to = 1855.1.1
        monthly_chance = 6
    }

    trigger = {
        OR = {
            always = yes
            has_dlc = "0_5_dawn_of_the_new_era"
        }
        stability < 25
    }

    illustration_tags = {
        10 = exterior
        10 = angry
    }

    immediate = {
        event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:peasants_estate }
    }

    option = {
        # 以改革平息
        name = flavor_hab_imperialism.1.a
        historical_option = yes
        add_stability = stability_mild_bonus
        add_estate_satisfaction = { type = estate_type:peasants_estate value = estate_satisfaction_mild_bonus }
        change_societal_value = { type = serfdom_vs_free_subjects value = societal_value_tiny_move_to_right }
        add_country_modifier = { modifier = habsburg_reform_era years = 15 }
    }

    option = {
        # 以军队镇压
        name = flavor_hab_imperialism.1.b
        add_stability = stability_severe_penalty
        add_country_modifier = { modifier = habsburg_military_absolutism years = 15 }
        add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_mild_bonus }
    }
}

            kill_character_silently = scope:viennese_composer

    }
