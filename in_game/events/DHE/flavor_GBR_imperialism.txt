# 维多利亚时代的帝国
flavor_gbr_imperialism.1 = {
    type = country_event
    fire_only_once = yes

    desc = flavor_gbr_imperialism.1.desc
    outcome = neutral

    historical_info = flavor_gbr_imperialism.1.historical_info

    dynamic_historical_event = {
        tag = ENG
        tag = GBR
        from = 1837.6.20
        to = 1901.1.1
        monthly_chance = 4
    }

    trigger = {
        OR = {
            is_colonial_overlord = yes
            any_subject = { is_subject_type = colonial_nation }
        }
    }

        10 = interior
        10 = happy
    }

    immediate = {
        event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:nobles_estate }
    }

    option = {
        name = flavor_gbr_imperialism.1.a
        historical_option = yes
        add_country_modifier = { modifier = victorian_imperial_zenith years = -1 }
        add_prestige = prestige_severe_bonus

        name = flavor_gbr_imperialism.1.b

# 英国文学与科学
flavor_gbr_imperialism.10 = {
    hide_portraits = yes
    type = country_event
    fire_only_once = yes
    title = flavor_gbr_imperialism.10.title
    desc = flavor_gbr_imperialism.10.desc
    outcome = neutral

    historical_info = flavor_gbr_imperialism.10.historical_info

    dynamic_historical_event = {
        tag = ENG
        tag = GBR
        from = 1840.1.1
        to = 1910.1.1
        monthly_chance = 2
    }

    trigger = {
        OR = {
            always = yes
            has_dlc = "0_5_dawn_of_the_new_era"
        }
        current_age_or_later = { age = age_7_imperialism }
    }

    illustration_tags = {
        10 = happy
        10 = interior
    }

    immediate = {
        event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:burghers_estate }
        create_character = {
            adm = { 30 90 }
            dip = { 30 90 }
            mil = { 10 50 }
            artist = artist_type:writer
            artist_skill = { 0.5 1.0 }
            birth_date = 1812.2.7
            female = no
            culture = root.culture
            religion = root.religion
            estate = estate_type:burghers_estate
            create_in_limbo = yes
            save_scope_as = british_writer
        }
    }

    option = {
        name = flavor_gbr_imperialism.10.a
        historical_option = yes
        scope:british_writer = {
            move_country = root
            add_character_modifier = {
                modifier = british_man_of_letters
                years = -1
            }
        }
        culture ?= {
            add_cultural_influence = cultural_influence_mild_bonus
            add_cultural_tradition = cultural_tradition_mild_bonus
        }
        add_research_progress = research_progress_mild_bonus
    }

    option = {
        name = flavor_gbr_imperialism.10.b
        add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_mild_bonus }
        change_societal_value = { type = traditionalist_vs_innovative value = societal_value_tiny_move_to_left }
    }
}
