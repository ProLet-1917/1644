namespace = lmc_situation_dynamic

lmc_situation_dynamic.1 = { #大同反正
    title = lmc_situation_dynamic.1.title
    desc = lmc_situation_dynamic.1.desc
    historical_info = lmc_situation_dynamic.1.historical_info
    fire_only_once = yes
    dynamic_historical_event = {
 		tag = CSH
 		from = 1644.5.1		# The game will check if this event can be fired after this date
 		to = 1700.1.1		# The game will stop checking after this date.
 		monthly_chance = 100	# Monthly chance that this event will fire (10 = 10% chance)
 	}
	trigger = {
		AND = {
            c:CJX ?= { is_subject_of = c:CSH }
			c:QNG ?= { has_variable = chi_domination }
            c:CSH ?= { 
                has_variable = chi_domination 
                chi_domination < 70
                NOT = { owns = location:funing }
            }
            c:WSG ?= {
                NOT = {
                    is_subject_of = c:CSH
                }
            }
            OR = {
                c:QNG.var:chi_domination > c:CSH.var:chi_domination
                c:MNG.var:chi_domination > c:CSH.var:chi_domination
            }
		}
	}
    immediate = {
        c:CJX = {
            set_variable = cjx_rebellion_first
        }
    }
    option = {#降清
        name = lmc_situation_dynamic.1.a
        trigger = {
            c:QNG.var:chi_domination > c:CSH.var:chi_domination
            c:QNG.var:chi_domination > c:MNG.var:chi_domination
        }
        c:CSH = {
            cancel_subject = c:CJX
        }
        c:CJX = {
            make_subject_of = {
                target = c:QNG
                type = subject_type:cn_march_governor
            }
        }
    }
    option = {#归明
        name = lmc_situation_dynamic.1.b
        trigger = {
            c:MNG.var:chi_domination > c:CSH.var:chi_domination
            c:MNG.var:chi_domination > c:QNG.var:chi_domination
        }
        c:CSH = {
            cancel_subject = c:CJX
        }
        c:CJX = {
            make_subject_of = {
                target = c:MNG
                type = subject_type:cn_march_governor
            }
        }
    }
}
