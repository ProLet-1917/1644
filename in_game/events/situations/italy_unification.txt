# Alliance members: annex into Italy
italy_unification.1001 = {
    type = country_event
    title = italy_unification.1001.title
    desc = italy_unification.1001.desc

    illustration_tags = {
        10 = interior
        10 = happy
    }

	trigger = {
		OR = {
		    always = yes
		    has_dlc = "0_5_dawn_of_the_new_era"
		}
		is_member_of_international_organization_of_type = { type = italy_unification_union }
	}

    immediate = {
        event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:burghers_estate }
        if = {
            limit = { exists = c:ITA }
            c:ITA = {
                save_scope_as = target_country
                ruler_or_regent ?= { save_scope_as = target_character }
            }
        }
    }

    option = {
        name = italy_unification.1001.a
        historical_option = yes
        ai_will_select = {
            value = 100
            add = { value = "opinion(scope:target_country)" }
        }
        add_prestige = prestige_severe_bonus
        add_stability = stability_mild_bonus
        if = {
            limit = {
                exists = scope:target_country
                scope:target_country = root
            }
            custom_tooltip = demand_vassalization_from_members_tt
        }
        else_if = {
            limit = {
                exists = scope:target_country
                is_subject = no
                NOT = { this = scope:target_country }
            }
            scope:target_country = {
                annex_country = {
                    country = root
                }
            }
        }
    }
}

italy_unification.1400 = {
    type = country_event
    title = italy_unification.1400.title
    desc = italy_unification.1400.desc

    historical_info = italy_unification.1400.historical_info
    major = yes

    illustration_tags = {
        10 = exterior
        10 = happy
    }

	trigger = {
		OR = {
		    always = yes
		    has_dlc = "0_5_dawn_of_the_new_era"
		}
		capital ?= { region = region:italy_region }
	}

    immediate = {
        event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:peasants_estate }
        create_character = {
            adm = { 30 70 }
            dip = { 50 90 }
            mil = { 70 100 }
            birth_date = 1807.7.4
            female = no
            culture = culture:ligurian
            religion = religion:catholic
            estate = estate_type:nobles_estate
            create_in_limbo = yes
            save_scope_as = garibaldi
        }
    }

    option = {
        name = italy_unification.1400.a
        historical_option = yes
        ai_will_select = {
            value = 100
        }
        scope:garibaldi = {
            move_country = root
        }
        add_country_modifier = {
            modifier = expedition_of_the_thousand
            years = 10
        }
        add_prestige = prestige_mild_bonus
    }

    option = {
        name = italy_unification.1400.b
        ai_will_select = {
            value = 20
        }
        hidden_effect = {
            kill_character_silently = scope:garibaldi
        }
        add_stability = stability_mild_bonus
    }
}
