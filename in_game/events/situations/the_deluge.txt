the_deluge.7 = { #Message from warsaw - Prussia for army
    type = country_event
    title = "the_deluge.7.t"
    desc = "the_deluge.7.desc"

    option = { #We accept
        name = "the_deluge.7.a"
        historical_option = yes
        c:PRU = {
            if = {
                limit = {
                    "this.war_with_country(c:ZAZ)" = yes
                }
                leave_war = {
                    actor = c:PLC
                    war = "c:ZAZ.war_with_country(c:PLC)"
                }
            }
        }
        c:PLC = {
            cancel_subject = c:PRU
        }
        c:PRU = {
            make_subject_of = {
                target = c:BRA
                type = subject_type:dominion
            }
        }
        location:berlin = {
            while = {
                count = 2
                create_sub_unit_with_owner = {
                    type = a_pikemen
                    owner = c:PLC
                    origin = this
                    experience = 35
                }
            }
            while = {
                count = 1
                create_sub_unit_with_owner = {
                    type = a_heavy_lancers
                    owner = c:PLC
                    origin = this
                    experience = 35
                }
            }
            while = {
                count = 1
                create_sub_unit_with_owner = {
                    type = a_chambered_cannon
                    owner = c:PLC
                    origin = this
                    experience = 35
                }
            }
        }
    }

    option = { #No
        name = "the_deluge.7.b"
        c:PRU = {
            remove_variable = already_declare_independence
        }
    }
}
