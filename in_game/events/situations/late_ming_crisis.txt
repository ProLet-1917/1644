namespace = late_ming_crisis
late_ming_crisis.1 = {
	#大同反正
	type = country_event
    category = situation_event
	title = late_ming_crisis.1.title
	desc = late_ming_crisis.1.desc
	#historical_info = late_ming_crisis.1.historical_info
	trigger = {
		c:CJX ?= {
			OR = {
				is_subject_of = c:CSH
				is_subject = no
			}
		}
		c:CSH ?= {
			NOT = {
				owns = location:funing
			}			#失去山海关
		}
		c:WSG ?= {
			NOT = {
				is_subject_of = c:CSH				#吴三桂不是大顺附庸
			}
		}
	}
	immediate = {
	}
	fire_only_once = yes
	option = {
		#降清
		name = late_ming_crisis.1.a
		cancel_subject = c:CJX
		historical_option = yes
		c:CJX = {
			make_subject_of = {
				target = c:QNG
				type = subject_type:cn_march_governor
			}
			# 让CJX加入大清的战争
			subject_join_overlord_wars = yes
		}
		# 增加CJX对大清的100点临时忠诚度
		c:CJX = {
			add_country_modifier = {
				modifier = loyalty_tifayifu
				years = 2
			}
		}
		c:CJX = {
			set_variable = cjx_rebellion_first
		}
		c:CSH = { 
			trigger_event_silently = {#通知大顺
				id = late_ming_crisis.2
				days = { 3 10 }
			} 
		}
		c:QNG = { 
			trigger_event_silently = {#通知大清
				id = qng_1644_unification.9
				days = { 5 25 }
			} 
		}
		kill_character = character:chi_zhang_tianlin
		 trigger = {
		     c:QNG.var:chi_domination > c:CSH.var:chi_domination
		     c:QNG.var:chi_domination > c:MNG.var:chi_domination
		}

	}
	option = {#归明
	     name = late_ming_crisis.1.b
	     trigger = {
	         c:MNG.var:chi_domination > c:CSH.var:chi_domination
	         c:MNG.var:chi_domination > c:QNG.var:chi_domination
	     }
	     c:CSH = {
	         cancel_subject = c:CJX
	     }
	     c:CJX = {
	         make_subject_of = {
	             target = c:MNG
	             type = subject_type:cn_march_governor
	         }
	     }
	}
	option = {
		#忠诚 
		name = late_ming_crisis.1.c
		ai_chance = 0
		c:CJX = {
            set_variable = cjx_refus_rebellion_first
			add_country_modifier = {
				modifier = loyalty_tifayifu
				years = 2
			}
		}
	}
}

late_ming_crisis.4 = { #河北山东叛乱
 	title = late_ming_crisis.4.title
 	desc = late_ming_crisis.4.desc
 	fire_only_once = yes
	trigger = {
		c:CSH ?= {
			owns = location:lingzhou_dezhou
			owns = location:licheng
		}
	}
	immediate = {
		set_global_variable = csh_rebellion_of_hebe_shandong
		c:CSH = {
			change_variable = {
				name = chi_mandate
				subtract = 5
			}
		}
		create_character = {#朱帅钦
			first_name = name_shuaiqin
			dynasty = dynasty:rtr_ming_zhu_dynasty
			religion = religion:sanjiao
			culture = culture:han_culture
			age = { 20 60 }
			female = no
			estate = estate_type:nobles_estate
			script = chi_zhu_shuaiqin
		}
		create_character = {#谢升
			first_name = name_sheng1_5347
			last_name = name_xie4_8b1d
			religion = religion:sanjiao
			culture = culture:han_culture
			birth_date = 1579.1.1
			birth_location = location:lingzhou_dezhou
			female = no
			estate = estate_type:nobles_estate
			script = chi_xie_sheng
			adm = 90
			dip = 85
		}
		ji_list_add = yes
		every_in_list = {
			list = list_of_ji_locations
			limit = { owner = c:CSH }
			add_core = c:CJI
		}
		c:CSH = {
			create_country_from_cores_in_our_locations = c:CJI
		}
		c:CJI = {
			set_new_ruler = character:chi_zhu_shuaiqin
			set_capital = location:lingzhou_dezhou
			change_country_flag = CHI_Ming_Sanchen
			set_country_rank = country_rank:rank_kingdom
			change_religion = religion:sanjiao
		}
		character:chi_xie_sheng = {
			move_country = c:CJI
		}
	}
 	option = {
 		name = late_ming_crisis.4.a	
		c:MNG = {#通知大明
			trigger_event_silently = {
				id = mng_1644.6
				days = { 10 30 }
			}
		}
 	}
}
