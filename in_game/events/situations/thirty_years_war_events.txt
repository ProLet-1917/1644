thirty_years_war_events.1 = { #Historical Peace of Westphalia
    type = country_event
    category = situation_event

    illustration_tags = {
        10 = regular
        10 = interior
    }

        east_pomerania_area = yes

        amberg_area = yes
        halberstadt_area = yes
        c:HAB = { white_peace = "war_with_country(c:HES)" }
    }

    option = {
        name = thirty_years_war_events.1.a
        situation:thirty_years_war = {
            end_situation = this
        }
        change_imperial_authority = { value = imperial_authority_unique_penalty }
        international_organization:hre = {
            add_policy_to_international_organization = policy:hre_religion_christian
            add_policy_to_international_organization = policy:only_imperial_religion_group_policy
            international_organization_add_special_status = {
                type = special_status:elector
                country = c:PAL
            }
            international_organization_chooses_new_leader = international_organization:hre
        }
        random_country = {
            destroy_international_organization = { target = international_organization:protestant_union }
            destroy_international_organization = { target = international_organization:catholic_league }
        }
        #c:KLE = {
        #    change_religion = religion:lutheran
        #}
        #international_organization:union_of_utrecht = {
        #    add_country_to_international_organization = c:KLE
        #}
        international_organization:swedish_empire = {
            add_country_to_international_organization = c:BRE
            international_organization_add_special_status = {
                type = special_status:dominion_of_empire
                country = c:BRE
            }
            international_organization_remove_special_status = {
                type = special_status:dominion_of_empire
                country = c:STE
            }
            international_organization_add_special_status = {
                type = special_status:crownland_of_swedish
                country = c:STE
            }
        }
        c:NED = { trigger_event_non_silently = netherland_independent.8 }
        c:STE = {
            make_subject_of = {
                target = c:SWE
                type = subject_type:fiefdom
            }
        }
        every_in_list = {
            list = east_pomerania_area_list
            limit = { owner = c:STE }
            change_location_owner = c:BRA
            change_control = 0.35
        }
        every_in_list = {
            list = gelre_area_list
            limit = { owner = c:KLE }
            change_location_owner = c:BRA
            change_control = 0.2
        }
        every_in_list = {
            list = halberstadt_area_list
            limit = { owner = c:MAG }
            change_location_owner = c:BRA
            change_control = 0.25
        }
        every_in_list = {
            list = wismar_area_list
            limit = { owner = c:MKL }
            change_location_owner = c:SWE
            change_control = 0.15
        }
        every_in_list = {
            list = jamtland_area_list
            limit = { owner = c:NOR }
            change_location_owner = c:SWE
            change_control = 0.15
        }
        every_in_list = {
            list = upper_alsace_area_list
            limit = { owner = c:TIR }
            change_location_owner = c:FRA
            change_control = 0.2
        }
        every_in_list = {
            list = amberg_area_list
            limit = { owner = c:UPA }
            change_location_owner = c:BAV
            change_control = 0.35
        }
        c:SPA = {
            declare_war_with_cb = {
                target = c:CAT
                type = casus_belli:cb_conquer_province
                target_province = c:CAT.province:barcelona_province
            }
        }
        every_country = {
            limit = {
                has_country_modifier = ai_in_religion_war
            }
            remove_country_modifier = ai_in_religion_war
        }
        if = {
            limit = {
                any_owned_location = {
                    owner = {
                        OR = {
                            this = c:NED
                            is_subject_or_below_of = c:NED
                        }
                    }
                }
            }
            save_scope_as = target_remove_location
        }
        international_organization:hre = {
            remove_location_from_international_organization = scope:target_remove_location
        }
    }
}
