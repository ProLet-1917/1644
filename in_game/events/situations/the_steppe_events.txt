the_steppe.2 = {
	type = country_event
	category = situation_event

	title = the_steppe.2.title
	desc = the_steppe.2.desc

	immediate = {
		ruler_or_regent ?= { save_scope_as = target_character1 }
		scope:target.owner = {
			save_scope_as = target_country
			ruler_or_regent ?= { save_scope_as = target_character2 }
		}
	}

	option = {
		name = the_steppe.2.a
		add_casus_belli = { target = scope:target.owner type = casus_belli:cb_conquer_another_horde province = scope:target }
		refresh_map_colors = yes
	}
}
