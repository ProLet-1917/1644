# Union members: annex into Germany
germany_unification.1001 = {
    type = country_event
    title = germany_unification.1001.title
    desc = germany_unification.1001.desc

    illustration_tags = {
        10 = interior
        10 = happy
    }

    immediate = {
        event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:burghers_estate }
        if = {
            limit = { exists = c:GER }
            c:GER = {
                save_scope_as = target_country
                ruler_or_regent ?= { save_scope_as = target_character }
            }
        }
    }

    option = {
        name = germany_unification.1001.a
        ai_will_select = {
            add = 100
            add = { value = "opinion(scope:target_country)" }
        }
        add_prestige = prestige_severe_bonus
        add_stability = stability_mild_bonus
        if = {
            limit = {
                exists = scope:target_country
                scope:target_country = root
            }
            custom_tooltip = demand_vassalization_from_members_tt
        }
        else_if = {
            limit = {
                exists = scope:target_country
                is_subject = no
                NOT = { this = scope:target_country }
            }
            scope:target_country = {
                annex_country = {
                    country = root
                }
            }
        }
    }
}
