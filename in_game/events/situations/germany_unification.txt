# Union members: annex into Germany
germany_unification.1001 = {
    type = country_event
    title = germany_unification.1001.title
    desc = germany_unification.1001.desc

    illustration_tags = {
        10 = interior
        10 = happy
    }

	trigger = {
		OR = {
		    always = yes
		    has_dlc = "0_5_dawn_of_the_new_era"
		}
		is_member_of_international_organization_of_type = { type = germany_unification_union }
	}

    immediate = {
        event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:burghers_estate }
        if = {
            limit = { exists = c:GER }
            c:GER = {
                save_scope_as = target_country
                ruler_or_regent ?= { save_scope_as = target_character }
            }
        }
    }

    option = {
        name = germany_unification.1001.a
        ai_will_select = {
            add = 100
            add = { value = "opinion(scope:target_country)" }
        }
        add_prestige = prestige_severe_bonus
        add_stability = stability_mild_bonus
        if = {
            limit = {
                exists = scope:target_country
                scope:target_country = root
            }
            custom_tooltip = demand_vassalization_from_members_tt
        }
        else_if = {
            limit = {
                exists = scope:target_country
                is_subject = no
                NOT = { this = scope:target_country }
            }
            scope:target_country = {
                annex_country = {
                    country = root
                }
            }
        }
    }
}

germany_unification.1200 = {
    type = country_event
    title = germany_unification.1200.title
    desc = germany_unification.1200.desc

    historical_info = germany_unification.1200.historical_info
    major = yes

    illustration_tags = {
        10 = exterior
        10 = angry
    }

	trigger = {
		OR = {
		    always = yes
		    has_dlc = "0_5_dawn_of_the_new_era"
		}
		culture = { has_culture_group = culture_group:german_group }
	}

    immediate = {
        event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:nobles_estate }
    }

    option = {
        # 以战争解决领导权
        name = germany_unification.1200.a
        historical_option = yes
        ai_will_select = {
            value = 60
        }
        add_country_modifier = {
            modifier = german_dual_leadership_contest
            years = 20
        }
        add_prestige = prestige_mild_penalty
        every_country = {
            limit = {
                has_variable = potential_unifier
                NOT = { this = root }
            }
            add_opinion = {
                target = root
                modifier = opinion_dishonored_alliance_defensive
            }
        }
    }

    option = {
        # 以议会与关税同盟解决
        name = germany_unification.1200.b
        ai_will_select = {
            value = 40
        }
        add_stability = stability_mild_bonus
        add_country_modifier = {
            modifier = german_dual_leadership_compromise
            years = 20
        }
    }
}
