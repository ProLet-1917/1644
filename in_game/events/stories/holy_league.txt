                name = unlock_ottomans_internal_decisions

}

holy_league.6 = {
    type = country_event
    title = holy_league.6.title
    desc = holy_league.6.desc
    fire_only_once = yes

    dynamic_historical_event = {
        tag = TUR
        from = 1674.1.1
        to = 1694.1.1
        monthly_chance = 5
    }

    trigger = {
        OR = {
            has_or_had_tag = TUR
            has_dlc = "0_4_echo_of_the_empire"
        }
        current_year >= 1674
        #OR = {
            c:HUN = {
                is_subject_of = c:HAB
            }
        #}
    }

    immediate = {
        event_illustration_estate_effect = { foreground = estate_type:peasants_estate background = estate_type:peasants_estate }
        c:HUN = {
            create_character = {
                estate = estate_type:nobles_estate
                first_name = name_emmerich.english_language
                dynasty = dynasty:thokoly_dynasty
                birth_date = 1657.9.25
                birth_location = location:kezhma
                culture = culture:hungarian
                religion = religion:lutheran
                script = emeric_thokoly
                adm = { 45 60 } dip = { 40 55 } mil = { 60 75 }
                save_scope_as = target_character
            }

            create_rebel = {
                category = estate
                estate = nobles_estate
                name = upper_hungary
                save_scope_as = upper_hungary
            }

            every_owned_location = {
                limit = {
                    OR = {
                        province_definition = province_definition:zemplen_province
                        province_definition = province_definition:szepes_province
                    }
                }
                every_pop = {
                    limit = {
                        owner = c:HUN
                        pop_type = pop_type:nobles
                    }
                    add_pop_satisfaction = pop_satisfaction_ultimate_penalty
                    change_pop_allegiance = scope:upper_hungary
                }
            }

            scope:emeric_thokoly = { change_character_allegiance = scope:upper_hungary }
            scope:upper_hungary = {
                start_civil_war = {
                    if = {
                        limit = {
                            root = { at_war = yes }
                        }
                        every_province = {
                            raise_levies = {
                                instant = yes
                                type = army
                            }
                        }
                    }
                }
            }

            scope:upper_hungary.capital = {
                while = {
                    count = 2
                    create_sub_unit_with_owner = {
                        type = a_heavy_lancers
                        owner = root
                        origin = this
                        experience = 75
                    }
                }
            }
        }
    }

    option = {
        historical_option = yes
        name = holy_league.6.a
        add_gold = -250
        if = {
            limit = {
                has_variable = unlock_ottomans_vienna_decisions
            }
            change_variable = {
                name = finished_ottomans_vienna_decisions
                add = 3 #To Do: Balance this
            }
        }
        scope:upper_hungary.capital = {
            while = {
                count = 1
                create_sub_unit_with_owner = {
                    type = a_heavy_lancers
                    owner = root
                    origin = this
                    experience = 75
                }
            }
            while = {
                count = 1
                create_sub_unit_with_owner = {
                    type = a_arquebusiers
                    owner = root
                    origin = this
                    experience = 65
                }
            }
        }
    }

    option = {
        name = holy_league.6.b

        if = {
            limit = {
                has_variable = unlock_ottomans_vienna_decisions
            }
            change_variable = {
                name = finished_ottomans_vienna_decisions
                add = 1
            }
        }
    }
}

    option = {
        historical_option = yes
        name = holy_league.8.a
        add_gold = -50
        if = {
            limit = {
                has_variable = finished_ottomans_vienna_decisions

                name = finished_ottomans_vienna_decisions

            }

                modifier = excellent_minister

                mode = add_and_extend

                limit = {
                    root = { country_can_promote_head_of_cabinet = yes }
                }
                custom_tooltip = {
                    text = becomes_head_of_cabinet_tt
                    root = {
                        promote_to_head_of_cabinet_effect = {
                            target = character:koprulu_mehmed

            }
        }
    }

    option = {
        name = holy_league.8.b
        add_gold = -25
        if = {
            limit = {
                has_variable = finished_ottomans_vienna_decisions
            }
