# Anno 1644 - Dynasty Stability
# Threshold events for the dynasty_stability variable (clamped -1000..1000) and the
# full dynastic cycle: decline, loss of the Mandate, change of dynasty and restoration.

namespace = dynasty_stability_1644

# 盛世
dynasty_stability_1644.1 = {
    type = country_event
    fire_only_once = yes
    title = dynasty_stability_1644.1.title
    desc = dynasty_stability_1644.1.desc
    outcome = positive

    trigger = {
        OR = {
            always = yes
            has_dlc = "0_5_dawn_of_the_new_era"
        }
        is_chinese_dynasty_leader = yes
        has_variable = dynasty_stability
        var:dynasty_stability >= 800
        NOT = { has_country_modifier = dynasty_golden_age }
    }

    illustration_tags = {
        10 = interior
        10 = happy
    }

    immediate = {
        event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:peasants_estate }
    }

    option = {
        name = dynasty_stability_1644.1.a
        add_country_modifier = { modifier = dynasty_golden_age years = -1 }
        add_prestige = prestige_severe_bonus
        add_legitimacy = legitimacy_mild_bonus
        every_subject = {
            limit = { is_subject_type = tributary }
            add_country_modifier = { modifier = imperial_favour years = 25 }
        }
    }
}

# 改朝换代
dynasty_stability_1644.5 = {
    type = country_event
    fire_only_once = yes
    title = dynasty_stability_1644.5.title
    desc = dynasty_stability_1644.5.desc
    outcome = negative

    major = yes

    trigger = {
        OR = {
            always = yes
            has_dlc = "0_5_dawn_of_the_new_era"
        }
        is_chinese_dynasty_leader = yes
        has_variable = dynasty_stability
        var:dynasty_stability <= -850
        NOT = { has_variable = new_dynasty_proclaimed }
    }

    illustration_tags = {
        10 = exterior
        10 = angry
    }

    immediate = {
        event_illustration_estate_effect = { foreground = estate_type:peasants_estate background = estate_type:nobles_estate }
        if = {
            limit = { has_ruler = yes }
            ruler ?= { save_scope_as = last_emperor }
        }
    }

    option = {
        # 禅让于新朝
        name = dynasty_stability_1644.5.a
        historical_option = yes
        ai_will_select = { base = 50 }
        hidden_effect = {
            if = {
                limit = { exists = scope:last_emperor }
                kill_character_silently = scope:last_emperor
            }
            set_variable = new_dynasty_proclaimed
        }
        add_country_modifier = { modifier = new_dynasty_consolidation years = 30 }
        change_dynasty_stability = { value = 500 }
        add_legitimacy = legitimacy_mild_penalty
        add_stability = stability_mild_penalty
        trigger_event_silently = { id = dynasty_stability_1644.6 days = 5 }
    }

    option = {
        # 死社稷
        name = dynasty_stability_1644.5.b
        ai_will_select = { base = 30 }
        hidden_effect = {
            if = {
                limit = { exists = scope:last_emperor }
                kill_character_silently = scope:last_emperor
            }
            set_variable = new_dynasty_proclaimed
        }
        add_country_modifier = { modifier = dynastic_collapse years = -1 }
        change_dynasty_stability = { value = 250 }
        add_stability = stability_extreme_penalty
        add_prestige = prestige_severe_penalty
        zzz_1644_disband_all_chinese_dynasties = yes
    }

    option = {
        # 中兴
        name = dynasty_stability_1644.5.c
        ai_will_select = { base = 20 }
        add_country_modifier = { modifier = dynastic_restoration years = 30 }
        change_dynasty_stability = { value = 300 }
        add_stability = stability_severe_penalty
        add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_severe_penalty }
    }
}
