# Anno 1644 - The Ming-Qing Transition (1644-1683)
#
# Continuation of the events already present in zzz_1644_mng.txt, zzz_1644_wsg_choice.txt,
# qing_1644_unification.txt and late_ming_crisis.txt. Only major, well-attested episodes
# are covered here; the 1644 opening is handled by the existing files.
#
# All events below are Dynamic Historical Events bound to a TAG. They fire on their own
# once the country exists and the date window is open - no on_action is required.

namespace = mqt_1644

# 剃发易服
mqt_1644.1 = {
    type = country_event
    fire_only_once = yes
    title = mqt_1644.1.title
    desc = mqt_1644.1.desc
    outcome = negative

    historical_info = mqt_1644.1.historical_info

    dynamic_historical_event = {
        tag = QNG
        from = 1644.6.1
        to = 1690.1.1
        monthly_chance = 40
    }

    trigger = {
        OR = {
            always = yes
            has_dlc = "0_5_dawn_of_the_new_era"
        }
        owns = location:dadu
        any_owned_location = {
            region = region:north_china_region
        }
    }

    illustration_tags = {
        10 = interior
        10 = angry
    }

    immediate = {
        event_illustration_estate_effect = { foreground = estate_type:peasants_estate background = estate_type:nobles_estate }
    }

    option = {
        # 严令剃发
        name = mqt_1644.1.a
        historical_option = yes
        ai_will_select = { base = 70 }
        add_country_modifier = { modifier = qing_queue_decree years = -1 }
        add_stability = stability_mild_penalty
        add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_mild_bonus }
        add_estate_satisfaction = { type = estate_type:peasants_estate value = estate_satisfaction_severe_penalty }
        change_societal_value = { type = sinicized_vs_unsinicized value = societal_value_tiny_move_to_right }
    }

    option = {
        # 暂缓剃发
        name = mqt_1644.1.b
        ai_will_select = { base = 30 }
        add_country_modifier = { modifier = qing_queue_deferred years = 20 }
        add_stability = stability_mild_bonus
        add_estate_satisfaction = { type = estate_type:peasants_estate value = estate_satisfaction_mild_bonus }
        change_societal_value = { type = sinicized_vs_unsinicized value = societal_value_tiny_move_to_left }
    }
}
