csh_1644_unification.2 = { #天翻地覆
    type = country_event
    title = csh_1644_unification.2.title
    desc = csh_1644_unification.2.desc
    historical_info = csh_1644_unification.2.historical_info
    immediate = {
        if = {
            limit = {
                has_variable = csh_attack_beijing
            }
            set_capital = location:dadu
            shandong_csh_list_add1 = yes
            trigger_event_silently = {
                id = csh_1644_unification.7
                days = 9
            }
        }
        shandong_csh_list_add2 = yes
        every_in_list = {
            list = list_of_shandong_csh
            limit = { owner = c:MNG }
            change_location_owner = c:CSH
            change_control = control_mild_bonus
        }
        c:MNG = {
            if = { #不南迁线
                limit = {
                    NOT = { has_variable = mng_move_to_nanjing }
                }
                if = {
                    limit = {
                        exists = character:chi_zhu_youjian
                    }
                    kill_character = character:chi_zhu_youjian
                }
                trigger_event_silently = {
                    id = mng_1644.2 #甲申之难
                }
                trigger_event_silently = {
                    id = mng_1644.5 #立新君
                    days = 10
                }
            }
            else = { #南迁线
                trigger_event_silently = {
                    id = mng_1644.3 #北都沦陷
                }
            }
        }
        c:WSG = { #没有南迁且不是吴三桂迎主的情况下吴三桂选择
            if = {
                limit = {
                    c:MNG = { NOT = { has_variable = mng_move_to_nanjing } }
                    NOT = { has_variable = wsg_1644_chosen_side }
                    c:CSH = { has_variable = csh_attack_beijing }
                }
                trigger_event_silently = {
                    id = wsg_1644_choice.1
                    days = 6
                }
            }
        }
        # 触发拷饷事件

    }
    option = {
        name = csh_1644_unification.2.a
        show_as_tooltip = {
            set_capital = location:dadu
            every_in_list = {
                list = shandong_csh_list_add1
                limit = { owner = c:MNG }
                change_location_owner = c:CSH
                change_control = control_mild_bonus
            }
        }
    }
}

csh_1644_unification.6 = { #山海关之战
    type = country_event
    title = csh_1644_unification.6.title
    desc = csh_1644_unification.6.desc
    historical_info = csh_1644_unification.6.historical_info
    
    
    immediate = {

        # 将北平地区的location添加到列表
        beiping_area_csh_list_add = yes
        
        # 失去北京和河北：将列表中的省份转移给QNG
        if = {
            limit = {
                exists = c:QNG
            }
            every_in_list = {
                list = list_of_beiping_csh
                limit = { owner = c:CSH }
                change_location_owner = c:QNG
                change_control = control_mild_penalty
            }
        }
        
        # 如果首都是北京，需要改变首都
        if = {
            limit = { capital = location:dadu }
            # 将首都改为西安
            set_capital = location:jingzhao
        }
        
        # 无论是否交战不利，都触发原有事件
        c:WSG = { trigger_event_silently = wsg_1644_choice.5 }
        c:QNG = { trigger_event_silently = qng_1644_unification.4 }
    }
    
    option = {
        name = csh_1644_unification.6.a
    }
}

csh_1644_unification.7 = { #在京铐掠
    type = country_event
    title = csh_1644_unification.7.title
    desc = csh_1644_unification.7.desc
    historical_info = csh_1644_unification.7.historical_info

    trigger = {
        tag = CSH
        has_variable = csh_attack_beijing
    }

    immediate = {
        ruler_or_regent = { save_scope_as = target_ruler }
        # 保存首都省份以便后续效果
        capital = {
            province = {
                save_scope_as = target_province
            }
        }
    }

    option = { #进行拷饷（史实选项）
        name = csh_1644_unification.7.a
        historical_option = yes

        add_gold = 5000

        # 失去稳定度（拷饷导致人心离散）
        #add_stability = stability_extreme_penalty
        c:CSH = {
            if = {
                limit = {
                    AND = {
                        has_variable = chi_domination
                        has_variable = chi_mandate
                    }
                }
                change_variable = {
				    name = chi_domination
				    subtract = 0
			    }
			    change_variable = {
				    name = chi_mandate
				    subtract = 10
			    }
		    }
        }

        #贵族不满
        add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_mild_penalty }

        # 首都省份的民众满意度下降和控制度下降（拷饷导致社会秩序混乱）
        scope:target_province = {
            every_location_in_province = {
                change_control = control_extreme_penalty
            }
        }

        # 设置变量，可能影响后续事件
        set_variable = csh_extortion_policy
    }

    option = { #不进行拷饷
        name = csh_1644_unification.7.b

        # 失去庶民的王朝改革
        if = {
            limit = {
                has_reform = government_reform:chi_peasant_empire_reform
            }
            remove_reform = government_reform:chi_peasant_empire_reform
        }

        add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_weak_bonus }

        # 增加战争疲劳（军饷不足导致士气下降）
        add_war_exhaustion = war_exhaustion_mild_bonus

        # 保持稳定度
        #add_stability = stability_weak_bonus

        # 首都省份的民众满意度略有提升和控制度提升（政策温和）
        scope:target_province = {
            every_location_in_province = {
                change_control = control_mild_bonus
            }
        }

        # 设置变量，表示采取了温和政策
        set_variable = csh_moderate_policy

    }
}
