namespace = dhe_shun_1644

# 均田免赋 - Land to the Tillers, Tax Remitted
dhe_shun_1644.1 = {
	type = country_event
	fire_only_once = yes
	title = dhe_shun_1644.1.title
	desc = dhe_shun_1644.1.desc
	outcome = neutral

	historical_info = dhe_shun_1644.1.historical_info

	dynamic_historical_event = {
		tag = CSH
		from = 1644.1.1
		to = 1652.1.1
		monthly_chance = 7
	}

	trigger = {
		OR = {
			always = yes
			has_dlc = "0_5_dawn_of_the_new_era"
		}
		tag = CSH
		is_subject = no
		current_year < 1652
	}

	illustration_tags = {
		10 = interior
		10 = happy
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:peasants_estate background = estate_type:peasants_estate }
	}

	option = {
		# 履亩而税，三年免征
		name = dhe_shun_1644.1.a
		historical_option = yes
		add_stability = stability_mild_bonus
		add_estate_satisfaction = { type = estate_type:peasants_estate value = estate_satisfaction_severe_bonus }
		add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_severe_penalty }
		change_societal_value = { type = serfdom_vs_free_subjects value = societal_value_tiny_move_to_right }
	}

	option = {
		# 先安军食
		name = dhe_shun_1644.1.b
		add_gold = 150
		add_manpower = { value = root.monthly_manpower multiply = 3 }
		add_estate_satisfaction = { type = estate_type:peasants_estate value = estate_satisfaction_severe_penalty }
		add_stability = stability_mild_penalty
	}
}

# 李自成之死 - The Death of Li Zicheng
dhe_shun_1644.3 = {
	type = country_event
	fire_only_once = yes
	title = dhe_shun_1644.3.title
	desc = dhe_shun_1644.3.desc
	outcome = negative

	major = yes

	historical_info = dhe_shun_1644.3.historical_info

	dynamic_historical_event = {
		tag = CSH
		from = 1645.1.1
		to = 1655.1.1
		monthly_chance = 3
	}

	trigger = {
		OR = {
			always = yes
			has_dlc = "0_5_dawn_of_the_new_era"
		}
		tag = CSH
		current_year >= 1645
		stability < 40
	}

	illustration_tags = {
		10 = exterior
		10 = angry
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:peasants_estate background = estate_type:nobles_estate }
		if = {
			limit = { has_ruler = yes }
			ruler ?= { save_scope_as = shun_leader }
		}
	}

	option = {
		# 秘不发丧
		name = dhe_shun_1644.3.a
		add_stability = stability_severe_penalty
		add_legitimacy = legitimacy_severe_penalty
		if = {
			limit = { exists = scope:shun_leader }
			kill_character_silently = scope:shun_leader
		}
		add_country_modifier = { modifier = shun_leadership_crisis years = 20 }
	}

	option = {
		# 推举新主，联明抗清
		name = dhe_shun_1644.3.b
		add_stability = stability_mild_penalty
		if = {
			limit = { exists = scope:shun_leader }
			kill_character_silently = scope:shun_leader
		}
		add_army_tradition = army_tradition_mild_bonus
		if = {
			limit = { country_exists = c:MNG }
			add_opinion = { target = c:MNG modifier = opinion_alliance }
		}
	}
}
